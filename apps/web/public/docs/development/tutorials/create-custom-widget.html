Create Custom Widget

Learn how to build reusable custom widgets for WebCraft

What You'll Build

In this tutorial, you'll create a custom "Testimonial Card" widget that displays customer testimonials with avatars, ratings, and animations. This widget will be reusable across different projects and configurable through the WebCraft interface.

Prerequisites

  • ✅ Basic understanding of React and TypeScript
  • ✅ WebCraft development environment set up

Step 2: Implement the Testimonial Card Component

Create the Main Component

// TestimonialCard.tsx
import React from 'react';
import { TestimonialCardProps } from './TestimonialCard.types';
import styles from './TestimonialCard.module.css';

export const TestimonialCard: React.FC = ({
  avatar,
  avatarInitials,
  name,
  title,
  company,
  rating,
  testimonial,
  theme = 'light',
  animation = 'fade',
  className = '',
  style,
  onClick
}) => {
  const [isVisible, setIsVisible] = React.useState(false);
  
  React.useEffect(() => {
    // Trigger animation on mount
    const timer = setTimeout(() => setIsVisible(true), 100);
    return () => clearTimeout(timer);
  }, []);

  const getAvatarContent = () => {
    if (avatar) {
      return (
        <img 
          src={avatar} 
          alt={name}
          className={styles.avatarImage}
          onError={(e) => {
            // Fallback to initials if image fails to load
            e.currentTarget.style.display = 'none';
          }}
        />
      );
    }
    
    if (avatarInitials) {
      return (
        <div className={styles.avatarInitials}>
          {avatarInitials}
        </div>
      );
    }
    
    // Default to first letter of name
    return (
      <div className={styles.avatarInitials}>
        {name.charAt(0).toUpperCase()}
      </div>
    );
  };

  const renderStars = () => {
    return Array.from({ length: 5 }, (_, index) => (
      <span 
        key={index}
        className={`${styles.star} ${index < rating ? styles.starFilled : styles.starEmpty}`}
      >
        ★
      </span>
    ));
  };

  const getAnimationClass = () => {
    if (!isVisible) return styles.hidden;
    
    switch (animation) {
      case 'slide':
        return styles.slideIn;
      case 'zoom':
        return styles.zoomIn;
      default:
        return styles.fadeIn;
    }
  };

  return (
    <div 
      className={`
        ${styles.testimonialCard}
        ${styles[theme]}
        ${getAnimationClass()}
        ${className}
      `}
      style={style}
      onClick={onClick}
    >
      <div className={styles.header}>
        <div className={styles.avatar}>
          {getAvatarContent()}
        </div>
        <div className={styles.authorInfo}>
          <h4 className={styles.name}>{name}</h4>
          {(title || company) && (
            <p className={styles.title}>
              {title}{title && company && ', '}{company}
            </p>
          )}
        </div>
      </div>
      
      <div className={styles.rating}>
        {renderStars()}
      </div>
      
      <blockquote className={styles.testimonial}>
        "{testimonial}"
      </blockquote>
    </div>
  );
};

export default TestimonialCard;

Step 4: Create Widget Wrapper for WebCraft

Create the Main Widget File

// index.tsx
import React from 'react';
import { WidgetWrapper } from '@/components/widgets/WidgetWrapper';
import { TestimonialCard } from './TestimonialCard';
import { testimonialCardConfig } from './testimonial-card.config';
import { TestimonialCardProps } from './TestimonialCard.types';

interface WebCraftTestimonialCardProps extends TestimonialCardProps {
  // WebCraft-specific props
  isSelected?: boolean;
  isHovered?: boolean;
  onSelect?: () => void;
  onUpdate?: (props: Partial) => void;
}

export const WebCraftTestimonialCard: React.FC = (props) => {
  const {
    isSelected,
    isHovered,
    onSelect,
    onUpdate,
    ...testimonialProps
  } = props;

  const handleClick = (e: React.MouseEvent) => {
    e.stopPropagation();
    onSelect?.();
  };

  return (
    <WidgetWrapper
      widgetId="testimonial-card"
      isSelected={isSelected}
      isHovered={isHovered}
      onSelect={handleClick}
      resizeHandles={['se']}
      minWidth={300}
      minHeight={200}
    >
      <TestimonialCard
        {...testimonialProps}
        onClick={handleClick}
        style={{
          ...testimonialProps.style,
          outline: isSelected ? '2px solid #3498db' : 'none',
          outlineOffset: '2px'
        }}
      />
    </WidgetWrapper>
  );
};

// Register the widget with WebCraft
export const testimonialCardWidget = {
  ...testimonialCardConfig,
  component: WebCraftTestimonialCard,
  
  // Serialization methods for saving/loading
  serialize: (props: TestimonialCardProps) => {
    return {
      ...props,
      // Remove any computed or runtime properties
      style: undefined,
      className: undefined
    };
  },
  
  deserialize: (data: any): TestimonialCardProps => {
    return {
      ...testimonialCardConfig.defaultProps,
      ...data
    };
  },
  
  // Validation method
  validate: (props: TestimonialCardProps) => {
    const errors: string[] = [];
    
    if (!props.name || props.name.length < 2) {
      errors.push('Name must be at least 2 characters');
    }
    
    if (!props.testimonial || props.testimonial.length < 10) {
      errors.push('Testimonial must be at least 10 characters');
    }
    
    if (props.rating < 1 || props.rating > 5) {
      errors.push('Rating must be between 1 and 5');
    }
    
    return errors;
  }
};

export default WebCraftTestimonialCard;

import { widgets } from '@/widgets';

export const WidgetLibrary: React.FC = () => {
  const categories = {
    content: [] as string[],
    layout: [] as string[],
    form: [] as string[],
    media: [] as string[]
  };

  // Group widgets by category
  Object.entries(widgets).forEach(([type, widget]) => {
    categories[widget.category].push(type);
  });

  return (
    <div className="widget-library">
      <h3>Widgets</h3>
      
      {Object.entries(categories).map(([category, widgetTypes]) => (
        widgetTypes.length > 0 && (
          <div key={category} className="widget-category">
            <h4>{category.charAt(0).toUpperCase() + category.slice(1)}</h4>
            <div className="widget-list">
              {widgetTypes.map(widgetType => {
                const widget = widgets[widgetType as keyof typeof widgets];
                return (
                  <DraggableWidget
                    key={widgetType}
                    type={widgetType}
                    icon={widget.icon}
                    label={widget.displayName}
                    description={widget.description}
                  />
                );
              })}
            </div>
          </div>
        )
      ))}
    </div>
  );
};

Step 6: Add Advanced Features

Add Animation Variants

// Enhanced TestimonialCard.tsx with more animations
const getAnimationVariants = (animation: string) => {
  switch (animation) {
    case 'slide':
      return {
        hidden: { opacity: 0, y: 50 },
        visible: { 
          opacity: 1, 
          y: 0,
          transition: { duration: 0.6, ease: 'easeOut' }
        }
      };
    case 'zoom':
      return {
        hidden: { opacity: 0, scale: 0.8 },
        visible: { 
          opacity: 1, 
          scale: 1,
          transition: { duration: 0.5, ease: 'easeOut' }
        }
      };
    case 'bounce':
      return {
        hidden: { opacity: 0, y: 30 },
        visible: { 
          opacity: 1, 
          y: 0,
          transition: {
            type: 'spring',
            damping: 10,
            stiffness: 100
          }
        }
      };
    default:
      return {
        hidden: { opacity: 0 },
        visible: { 
          opacity: 1,
          transition: { duration: 0.6 }
        }
      };
  }
};

    </motion.div>
  );
};

Add Responsive Behavior

// Add responsive props
export interface TestimonialCardProps {
  // ... existing props
  responsive?: {
    mobile?: Partial;
    tablet?: Partial;
    desktop?: Partial;
  };
  breakpoints?: {
    mobile?: number;
    tablet?: number;
    desktop?: number;
  };
}

// Responsive hook
import { useMediaQuery } from 'react-responsive';

export const TestimonialCard: React.FC = (props) => {
  const isMobile = useMediaQuery({ maxWidth: 768 });
  const isTablet = useMediaQuery({ maxWidth: 1024, minWidth: 769 });
  const isDesktop = useMediaQuery({ minWidth: 1025 });

  // Get responsive props based on current breakpoint
  const getResponsiveProps = () => {
    if (isMobile && props.responsive?.mobile) {
      return { ...props, ...props.responsive.mobile };
    }
    if (isTablet && props.responsive?.tablet) {
      return { ...props, ...props.responsive.tablet };
    }
    if (isDesktop && props.responsive?.desktop) {
      return { ...props, ...props.responsive.desktop };
    }
    return props;
  };

  const responsiveProps = getResponsiveProps();

  return (
    // Use responsiveProps instead of props
    <motion.div {...responsiveProps}>
      {/* ... component content */}
    </motion.div>
  );
};

// TestimonialCard.test.tsx
import React from 'react';
import { render, screen, fireEvent } from '@testing-library/react';
import { TestimonialCard } from './TestimonialCard';

describe('TestimonialCard', () => {
  const defaultProps = {
    name: 'John Doe',
    title: 'CEO',
    company: 'TechCorp',
    rating: 5,
    testimonial: 'This is an amazing product!',
    theme: 'light' as const,
    animation: 'fade' as const
  };

  it('renders with all props', () => {
    render(<TestimonialCard {...defaultProps} />);
    
    expect(screen.getByText('John Doe')).toBeInTheDocument();
    expect(screen.getByText('CEO, TechCorp')).toBeInTheDocument();
    expect(screen.getByText('This is an amazing product!')).toBeInTheDocument();
    expect(screen.getAllByText('★').length).toBe(5);
  });

  it('renders with avatar image', () => {
    const avatarUrl = 'https://example.com/avatar.jpg';
    render(<TestimonialCard {...defaultProps} avatar={avatarUrl} />);
    
    const avatar = screen.getByRole('img');
    expect(avatar).toHaveAttribute('src', avatarUrl);
    expect(avatar).toHaveAttribute('alt', 'John Doe');
  });

  it('renders with avatar initials when no avatar provided', () => {
    render(<TestimonialCard {...defaultProps} avatarInitials="JD" />);
    
    expect(screen.getByText('JD')).toBeInTheDocument();
  });

  it('renders with default initials when no avatar or initials provided', () => {
    render(<TestimonialCard {...defaultProps} />);
    
    expect(screen.getByText('J')).toBeInTheDocument();
  });

  it('handles click events', () => {
    const handleClick = jest.fn();
    render(<TestimonialCard {...defaultProps} onClick={handleClick} />);
    
    fireEvent.click(screen.getByRole('article'));
    expect(handleClick).toHaveBeenCalledTimes(1);
  });

  it('applies theme classes correctly', () => {
    const { container, rerender } = render(<TestimonialCard {...defaultProps} theme="light" />);
    expect(container.firstChild).toHaveClass('light');
    
    rerender(<TestimonialCard {...defaultProps} theme="dark" />);
    expect(container.firstChild).toHaveClass('dark');
  });

  it('renders correct number of filled stars', () => {
    render(<TestimonialCard {...defaultProps} rating={3} />);
    
    const filledStars = screen.getAllByText('★').filter(star => 
      star.classList.contains('starFilled')
    );
    expect(filledStars.length).toBe(3);
  });
});

Best Practices

Widget Development Guidelines

  • Keep it Simple: Focus on one primary function per widget
  • Make it Configurable: Provide meaningful customization options
  • Ensure Responsiveness: Test on different screen sizes
  • Add Animations Thoughtfully: Use subtle animations that enhance UX
  • Document Thoroughly: Include clear descriptions and examples
  • Test Extensively: Write unit tests and test edge cases
  • Follow Naming Conventions: Use consistent naming patterns
  • Consider Accessibility: Add ARIA labels and keyboard support

Performance Optimization

  • Use React.memo for expensive components
  • Implement proper shouldComponentUpdate logic
  • Lazy load heavy assets like images
  • Optimize CSS with CSS modules
  • Use CSS-in-JS sparingly

Configuration Best Practices

  • Provide sensible defaults
  • Validate all inputs
  • Group related properties
  • Use appropriate input types
  • Add helpful descriptions
