Quick Diagnostics

Use these commands for rapid troubleshooting of common issues:

# System Health Check
#!/bin/bash
echo "=== WebCraft Health Check ==="

# Check if services are running
echo "Checking service status..."
docker-compose ps

# Check backend health
echo "Checking backend health..."
curl -f http://localhost:8000/health || echo "❌ Backend unhealthy"

# Check frontend
echo "Checking frontend..."
curl -f http://localhost:3000 || echo "❌ Frontend unavailable"

# Check database connection
echo "Checking database..."
docker-compose exec postgres pg_isready -U webcraft || echo "❌ Database connection failed"

# Check Redis
echo "Checking Redis..."
docker-compose exec redis redis-cli ping || echo "❌ Redis connection failed"

# Check disk space
echo "Checking disk space..."
df -h | grep -E "(Filesystem|/dev/)"

# Check memory usage
echo "Checking memory usage..."
free -h

# Check for recent errors
echo "Checking recent errors..."
docker-compose logs --tail=50 --timestamps backend | grep -i error || echo "No recent backend errors"
docker-compose logs --tail=50 --timestamps frontend | grep -i error || echo "No recent frontend errors"

echo "=== Health check complete ==="

Common Issues

Database Connection Issues

Symptoms:
  • Backend returns 500 errors
  • "Connection refused" errors in logs
  • "Database does not exist" errors
  • Timeout errors during startup
Diagnosis: 
# Check if PostgreSQL is running
docker-compose ps postgres

# Check PostgreSQL logs
docker-compose logs postgres

# Test connection manually
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "SELECT 1;"

# Check network connectivity
docker-compose exec backend nc -zv postgres 5432

# Check environment variables
docker-compose exec backend env | grep DATABASE
Solutions:
  1. PostgreSQL not running: 
    # Restart PostgreSQL
docker-compose restart postgres

# Check if port is already in use
netstat -tulpn | grep 5432

# Check PostgreSQL configuration
cat docker-compose.yml | grep postgres
  2. Database doesn't exist: 
    # Create database manually
docker-compose exec postgres createdb -U webcraft webcraft_dev

# Run migrations
alembic upgrade head

# Or use Docker Compose
docker-compose exec backend python -m alembic upgrade head
  3. Authentication failed: 
    # Check PostgreSQL users
docker-compose exec postgres psql -U postgres -c "\du"

# Reset PostgreSQL password
docker-compose exec postgres psql -U postgres -c "ALTER USER webcraft PASSWORD 'new_password';"

# Update connection string in .env file
  4. Network issues: 
    # Check Docker network
docker network ls
docker network inspect webcraft_default

# Recreate network
docker-compose down
docker network prune
docker-compose up -d

Redis Connection Issues

Symptoms:
  • Cache operations fail
  • Session management issues
  • "Connection refused" errors
  • High latency on cached operations
Diagnosis: 
# Check Redis status
docker-compose ps redis

# Check Redis logs
docker-compose logs redis

# Test Redis connection
docker-compose exec redis redis-cli ping

# Check Redis memory usage
docker-compose exec redis redis-cli info memory

# Check Redis configuration
docker-compose exec redis redis-cli config get maxmemory
Solutions:
  1. Redis not responding: 
    # Restart Redis
docker-compose restart redis

# Check Redis configuration
cat docker-compose.yml | grep redis

# Monitor Redis logs in real-time
docker-compose logs -f redis
  2. Memory issues: 
    # Check memory usage
docker-compose exec redis redis-cli info memory

# Clear all cache (use with caution)
docker-compose exec redis redis-cli FLUSHALL

# Set memory limit
docker-compose exec redis redis-cli config set maxmemory 256mb
docker-compose exec redis redis-cli config set maxmemory-policy allkeys-lru
  3. Connection pool exhaustion: 
    # Check connected clients
docker-compose exec redis redis-cli info clients

# Check connection limits
docker-compose exec redis redis-cli config get maxclients

# Monitor active connections
docker-compose exec redis redis-cli client list

Frontend Build Issues

Symptoms:
  • Build fails with module resolution errors
  • TypeScript compilation errors
  • Webpack bundling issues
  • Missing dependencies
Diagnosis: 
# Check Node.js version
node --version

# Check package.json
cat apps/frontend/package.json | grep -E "(node|npm)"

# Check for missing dependencies
npm ls

# Check TypeScript errors
npm run type-check

# Check linting errors
npm run lint

# Clear npm cache
npm cache clean --force
Solutions:
  1. Node.js version mismatch: 
    # Use correct Node.js version (check .nvmrc)
cd apps/frontend
nvm use

# Or install correct version
nvm install 18.17.0
nvm use 18.17.0
  2. Missing dependencies: 
    # Clean install
cd apps/frontend
rm -rf node_modules package-lock.json
npm install

# Check for peer dependency warnings
npm install 2>&1 | grep -i peer
  3. TypeScript errors: 
    # Check TypeScript configuration
cat apps/frontend/tsconfig.json

# Run TypeScript compiler directly
npx tsc --noEmit

# Skip type checking during build (temporary)
NEXT_BUILD_IGNORE_TYPE_CHECK=true npm run build
  4. Memory issues during build: 
    # Increase Node.js memory limit
export NODE_OPTIONS="--max-old-space-size=4096"
npm run build

# Build with verbose output
npm run build --verbose

API Authentication Issues

Symptoms:
  • 401 Unauthorized errors
  • JWT token validation failures
  • Session management problems
  • CORS errors during authentication
Diagnosis: 
# Check JWT configuration
docker-compose exec backend env | grep JWT

# Check authentication logs
docker-compose logs backend | grep -i auth

# Test authentication endpoint
curl -X POST http://localhost:8000/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"test@example.com","password":"password"}'

# Check CORS configuration
docker-compose exec backend env | grep CORS
Solutions:
  1. JWT secret mismatch: 
    # Check JWT secret in environment
docker-compose exec backend env | grep JWT_SECRET

# Verify JWT secret consistency across services
cat docker-compose.yml | grep JWT_SECRET

# Regenerate JWT secret (if needed)
openssl rand -hex 32
  2. CORS configuration issues: 
    # Check CORS origins
docker-compose exec backend env | grep CORS

# Test with curl including origin
curl -X POST http://localhost:8000/v1/auth/login \
  -H "Origin: http://localhost:3000" \
  -H "Content-Type: application/json" \
  -d '{"email":"test@example.com","password":"password"}'
  3. Token expiration issues: 
    # Check token expiration settings
docker-compose exec backend env | grep JWT_EXPIRATION

# Test token validation
curl -X GET http://localhost:8000/v1/user/me \
  -H "Authorization: Bearer YOUR_TOKEN_HERE"

Performance Issues

Symptoms:
  • High response times
  • Database query timeouts
  • Memory usage spikes
  • CPU usage at 100%
Diagnosis: 
# Check system resources
docker stats --no-stream

# Check slow queries in PostgreSQL
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT query, calls, total_time, mean_time 
FROM pg_stat_statements 
ORDER BY total_time DESC 
LIMIT 10;"

# Check Redis performance
docker-compose exec redis redis-cli info stats | grep keyspace

# Check application logs for slow requests
docker-compose logs backend | grep -E "(slow|timeout|took)"

# Monitor real-time logs
docker-compose logs -f backend | grep -E "(GET|POST|PUT|DELETE)"
Solutions:
  1. Database performance: 
    # Check for missing indexes
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT schemaname, tablename, attname, n_distinct, correlation 
FROM pg_stats 
WHERE schemaname = 'public' AND n_distinct > 100 
ORDER BY n_distinct DESC;"

# Check table sizes
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT schemaname, tablename, 
       pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) as size
FROM pg_tables 
WHERE schemaname = 'public' 
ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC;"

# Update table statistics
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "VACUUM ANALYZE;"
  2. Application performance: 
    # Check for memory leaks
docker stats --no-stream | grep webcraft

# Monitor garbage collection (if Python)
docker-compose exec backend python -c "import gc; print(gc.get_stats())"

# Check for connection leaks
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT count(*), state 
FROM pg_stat_activity 
GROUP BY state;"

# Enable query logging temporarily
docker-compose exec postgres psql -U postgres -c "
ALTER SYSTEM SET log_min_duration_statement = 1000;
SELECT pg_reload_conf();"
  3. Redis performance: 
    # Check Redis memory usage
docker-compose exec redis redis-cli info memory

# Check for large keys
docker-compose exec redis redis-cli --bigkeys

# Monitor Redis commands
docker-compose exec redis redis-cli monitor

FAQ - Frequently Asked Questions

Installation & Setup

Q: I'm getting "command not found" errors during installation. What should I do?

A: This usually means required tools aren't installed or not in your PATH.

# Check if required tools are installed
which docker docker-compose node npm python3 pip3

# Install missing tools
# Ubuntu/Debian:
sudo apt update
sudo apt install docker.io docker-compose nodejs npm python3 python3-pip

# macOS:
brew install docker docker-compose node python3

# Add user to docker group (Linux)
sudo usermod -aG docker $USER
newgrp docker

Q: Docker containers keep restarting. How do I fix this?

A: Check container logs to identify the cause of the restart.

# Check container status
docker-compose ps

# View logs for failing containers
docker-compose logs --tail=100 backend
docker-compose logs --tail=100 frontend
docker-compose logs --tail=100 postgres
docker-compose logs --tail=100 redis

# Common fixes:
# 1. Check port conflicts
netstat -tulpn | grep -E ":(3000|8000|5432|6379)"

# 2. Check disk space
df -h

# 3. Check memory usage
free -h

# 4. Restart all services
docker-compose down
docker-compose up -d

Q: How do I reset my development environment completely?

A: Follow these steps for a complete reset:

# Stop all services
docker-compose down

# Remove all containers and volumes
docker-compose down --volumes --remove-orphans

# Remove all images (optional)
docker-compose down --rmi all

# Clean up Docker system
docker system prune -a

# Remove node_modules (frontend)
rm -rf apps/frontend/node_modules

# Rebuild and start
docker-compose up --build -d

# Run migrations
alembic upgrade head

# Or using Docker
docker-compose exec backend alembic upgrade head

Development & Debugging

Q: How do I enable debug mode for better error messages?

A: Set the appropriate environment variables for your services.

# Backend debug mode
export ENVIRONMENT=development
export LOG_LEVEL=debug

# Frontend debug mode
export NODE_ENV=development
export NEXT_PUBLIC_DEBUG=true

# Update docker-compose.yml
# Add to backend service environment:
ENVIRONMENT: development

# Add to frontend service environment:
NODE_ENV: development
NEXT_PUBLIC_DEBUG: "true"

# Restart services
docker-compose restart backend frontend

Q: How do I perform a zero-downtime deployment?

A: Follow these steps for zero-downtime deployments:

# 1. Use rolling updates (configured in deployment.yaml)
kubectl apply -f k8s/backend-deployment.yaml

# 2. Monitor rollout status
kubectl rollout status deployment/webcraft-backend -n webcraft

# 3. Test new version before full rollout
kubectl set image deployment/webcraft-backend \
  backend=webcraft/backend:new-version \
  --record -n webcraft

# 4. If issues occur, rollback immediately
kubectl rollout undo deployment/webcraft-backend -n webcraft

# 5. Use readiness probes to ensure pods are ready
kubectl describe pod  -n webcraft | grep -A 5 "Readiness"

# 6. Gradual rollout with canary deployment
# Use Flagger or similar tool for advanced canary deployments

# 7. Blue-green deployment (alternative approach)
# Deploy to new environment, test, then switch traffic

# 8. Monitor during deployment
kubectl logs -f deployment/webcraft-backend -n webcraft | grep -E "(error|Error)"

# 9. Verify health after deployment
curl -f https://api.yourdomain.com/health

Database Issues

Q: How do I handle database migration failures?

A: Follow these steps to recover from migration failures:

# Check current migration status
alembic current
alembic history --verbose

# Check failed migrations
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT * FROM alembic_version;"

# If migration is stuck, check locks
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT * FROM pg_locks WHERE NOT granted;"

# Kill blocking connections if necessary
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT pg_terminate_backend(pid) 
FROM pg_stat_activity 
WHERE pid <> pg_backend_pid() 
AND state = 'idle in transaction';"

# Rollback to previous migration
alembic downgrade -1

# Or rollback to specific revision
alembic downgrade base

# Fix migration script and retry
# Edit the migration file in alembic/versions/
alembic upgrade head

# If all else fails, manual intervention
# Backup database first
docker-compose exec postgres pg_dump -U webcraft webcraft_dev > backup.sql

# Manually fix database state
# Then mark migration as applied
alembic stamp head

Q: How do I optimize slow database queries?

A: Use these techniques to identify and optimize slow queries:

# Enable query logging
docker-compose exec postgres psql -U postgres -c "
ALTER SYSTEM SET log_min_duration_statement = 1000;
SELECT pg_reload_conf();"

# Find slow queries
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT query, calls, total_time, mean_time, rows 
FROM pg_stat_statements 
ORDER BY total_time DESC 
LIMIT 10;"

# Analyze query execution plan
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
EXPLAIN ANALYZE SELECT * FROM users WHERE email = 'test@example.com';"

# Check for missing indexes
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT schemaname, tablename, attname, n_distinct 
FROM pg_stats 
WHERE schemaname = 'public' AND n_distinct > 100 
ORDER BY n_distinct DESC;"

# Create appropriate indexes
CREATE INDEX CONCURRENTLY idx_users_email ON users(email);
CREATE INDEX CONCURRENTLY idx_workspaces_user_id ON workspaces(user_id);

# Update table statistics
VACUUM ANALYZE users;

# Monitor query performance after optimization
docker-compose exec postgres psql -U webcraft -d webcraft_dev -c "
SELECT query, calls, total_time, mean_time 
FROM pg_stat_statements 
WHERE query LIKE '%users%' 
ORDER BY mean_time DESC;"

    # Development environment
docker-compose restart postgres redis

Security Incident Response

  1. Identify the threat: 
    # Check for suspicious activity
docker-compose logs | grep -E "(failed|unauthorized|invalid)"
kubectl logs --all-containers=true | grep -E "(failed|unauthorized|invalid)"

# Check for unusual traffic
kubectl logs deployment/webcraft-backend -n webcraft | grep -E "(rate.limit|too.many.requests)"

# Monitor authentication attempts
docker-compose logs backend | grep -E "(login.failed|invalid.credentials)"
  2. Isolate affected services: 
    # Scale down affected services
kubectl scale deployment webcraft-backend --replicas=0 -n webcraft

# Block suspicious IPs
kubectl apply -f - <
  3. Rotate compromised credentials: 
    # Generate new JWT secret
openssl rand -hex 32

# Update Kubernetes secrets
kubectl create secret generic webcraft-secrets \
  --from-literal=jwt-secret="$(openssl rand -hex 32)" \
  --dry-run=client -o yaml | kubectl apply -f -

# Rotate database passwords
# (Follow your cloud provider's password rotation process)

# Restart services to pick up new secrets
kubectl rollout restart deployment/webcraft-backend -n webcraft
  4. Document and report: 
    # Collect evidence
docker-compose logs --since=1h > incident-logs-$(date +%Y%m%d-%H%M).txt
