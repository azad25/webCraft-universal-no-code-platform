# main.py - Application factory
from fastapi import FastAPI
from contextlib import asynccontextmanager

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Initialize services on startup"""
    # Initialize database
    await init_db()
    
    # Initialize Redis
    await init_redis()
    
    # Initialize Kafka

    # Initialize module system

Router Registration

Modular router registration with automatic prefixing and middleware application.

# Router registration pattern
from routers import auth, apps, widgets, automations

def register_routers(app: FastAPI):
    """Register all API routers"""
    routers = [
        (auth.router, "/auth", ["Authentication"]),
        (apps.router, "/apps", ["Applications"]),
        (widgets.router, "/widgets", ["Widgets"]),
        (automations.router, "/automations", ["Automations"]),
        (integrations.router, "/integrations", ["Integrations"]),
        (ai.router, "/ai", ["AI Services"]),
    ]
    
    for router, prefix, tags in routers:
        app.include_router(
            router,
            prefix=f"/api/v1{prefix}",
            tags=tags
        )

# Register routers
register_routers(app)

Database Architecture

Multi-Tenant Architecture

WebCraft implements a multi-tenant database architecture with proper isolation and security.

# Multi-tenant database session
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import sessionmaker

class TenantSession:
    """Multi-tenant database session manager"""
    
    def __init__(self):
        self.engine = create_async_engine(
            settings.DATABASE_URL,
            echo=settings.DEBUG,
            pool_size=20,
            max_overflow=0,
        )
        
        self.async_session = sessionmaker(
            self.engine,
            class_=AsyncSession,
            expire_on_commit=False
        )
    
    async def get_session(self, tenant_id: str = None) -> AsyncSession:
        """Get database session with tenant isolation"""
        session = self.async_session()
        
        if tenant_id:
            # Set tenant context for queries
            await session.execute(
                text("SET app.current_tenant = :tenant_id"),
                {"tenant_id": tenant_id}
            )
        
        return session

# Dependency injection
db_session = TenantSession()

async def get_db(tenant_id: str = Depends(get_current_tenant)) -> AsyncSession:
    """Database dependency for FastAPI"""
    async with db_session.get_session(tenant_id) as session:
        yield session

from sqlalchemy.orm import relationship

class User(Base):
    __tablename__ = "users"
    
    id = Column(Integer, primary_key=True, index=True)
    email = Column(String, unique=True, index=True, nullable=False)
    username = Column(String, unique=True, index=True, nullable=False)
    hashed_password = Column(String, nullable=False)
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), onupdate=func.now())
    
    # Relationships
    apps = relationship("App", back_populates="owner", cascade="all, delete-orphan")
    integrations = relationship("Integration", back_populates="user")
    
    # Indexes
    __table_args__ = (
        Index('idx_user_email', 'email'),
        Index('idx_user_username', 'username'),
    )

class App(Base):
    __tablename__ = "apps"
    
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    slug = Column(String, unique=True, index=True, nullable=False)
    description = Column(Text)
    owner_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    
    # Relationships
    owner = relationship("User", back_populates="apps")
    widgets = relationship("Widget", back_populates="app", cascade="all, delete-orphan")
    pages = relationship("Page", back_populates="app", cascade="all, delete-orphan")
    
    # Methods
    def to_dict(self) -> dict:
        return {
            "id": self.id,
            "name": self.name,
            "slug": self.slug,
            "description": self.description,
            "owner_id": self.owner_id,
            "created_at": self.created_at.isoformat(),
        }

Service Layer Architecture

Service Base Class

Abstract base class for all services with common functionality.

# Base service class
from abc import ABC, abstractmethod
from typing import Any, Dict, Optional
from sqlalchemy.ext.asyncio import AsyncSession
from core.logging import get_logger

class BaseService(ABC):
    """Abstract base class for all services"""
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.logger = get_logger(self.__class__.__name__)
    
    @abstractmethod
    async def validate(self, data: Dict[str, Any]) -> bool:
        """Validate input data"""
        pass
    
    @abstractmethod
    async def execute(self, **kwargs) -> Any:
        """Execute service logic"""
        pass
    
    async def log_operation(self, operation: str, details: Dict[str, Any]):
        """Log service operations"""
        self.logger.info(f"{operation}: {details}")
    
    async def handle_error(self, error: Exception, context: Dict[str, Any]):
        """Handle service errors"""
        self.logger.error(f"Service error: {error}", extra=context)
        raise error

# Example service implementation
class AppBuilderService(BaseService):
    """Service for building and managing applications"""
    
    async def validate(self, data: Dict[str, Any]) -> bool:
        """Validate app creation data"""
        required_fields = ['name', 'owner_id']
        for field in required_fields:
            if field not in data:
                raise ValueError(f"Missing required field: {field}")
        return True
    
    async def execute(self, **kwargs) -> App:
        """Create a new application"""
        try:
            await self.validate(kwargs)
            
            app = App(
                name=kwargs['name'],
                slug=self.generate_slug(kwargs['name']),
                description=kwargs.get('description', ''),
                owner_id=kwargs['owner_id']
            )
            
            self.db.add(app)
            await self.db.commit()
            await self.db.refresh(app)
            
            await self.log_operation("app_created", {
                "app_id": app.id,
                "name": app.name,
                "owner_id": app.owner_id
            })
            
            return app
            
        except Exception as e:
            await self.handle_error(e, {"operation": "create_app", "data": kwargs})
    
    def generate_slug(self, name: str) -> str:
        """Generate URL-friendly slug from app name"""
        import re
        slug = re.sub(r'[^\w\s-]', '', name.lower())
        slug = re.sub(r'[-\s]+', '-', slug)
        return slug.strip('-')

Async Database Operations

Efficient async database operations with proper connection management.

# Async database operations
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, and_, or_

class AsyncRepository:
    """Generic async repository pattern"""
    
    def __init__(self, model: Type[Base], db: AsyncSession):
        self.model = model
        self.db = db
    
    async def create(self, **kwargs) -> Base:
        """Create a new record"""
        instance = self.model(**kwargs)
        self.db.add(instance)
        await self.db.commit()
        await self.db.refresh(instance)
        return instance
    
    async def get_by_id(self, id: int) -> Optional[Base]:
        """Get record by ID"""
        result = await self.db.execute(
            select(self.model).where(self.model.id == id)
        )
        return result.scalar_one_or_none()
    
    async def get_all(self, skip: int = 0, limit: int = 100) -> List[Base]:
        """Get all records with pagination"""
        result = await self.db.execute(
            select(self.model)
            .offset(skip)
            .limit(limit)
        )
        return result.scalars().all()
    
    async def update(self, id: int, **kwargs) -> Optional[Base]:
        """Update record by ID"""
        instance = await self.get_by_id(id)
        if instance:
            for key, value in kwargs.items():
                setattr(instance, key, value)
            await self.db.commit()
            await self.db.refresh(instance)
        return instance
    
    async def delete(self, id: int) -> bool:
        """Delete record by ID"""
        instance = await self.get_by_id(id)
        if instance:
            await self.db.delete(instance)
            await self.db.commit()
            return True
        return False

# Usage example
user_repo = AsyncRepository(User, db)
user = await user_repo.create(
    email="user@example.com",
    username="johndoe",
    hashed_password="hashed_password_here"
)

Concurrent API Calls

Efficient concurrent API calls using asyncio.gather for better performance.

# Concurrent API calls
import asyncio
import aiohttp
from typing import List, Dict, Any

class ConcurrentAPIClient:
    """Client for making concurrent API calls"""
    
    def __init__(self, max_concurrent: int = 10):
        self.max_concurrent = max_concurrent
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    async def fetch_single(self, session: aiohttp.ClientSession, url: str, **kwargs) -> Dict[str, Any]:
        """Fetch single URL with semaphore"""
        async with self.semaphore:
            try:
                async with session.get(url, **kwargs) as response:
                    return await response.json()
            except Exception as e:
                return {"error": str(e), "url": url}
    
    async def fetch_multiple(self, urls: List[str], **kwargs) -> List[Dict[str, Any]]:
        """Fetch multiple URLs concurrently"""
        async with aiohttp.ClientSession() as session:
            tasks = [
                self.fetch_single(session, url, **kwargs)
                for url in urls
            ]
            return await asyncio.gather(*tasks)
    
    async def fetch_with_retry(self, url: str, max_retries: int = 3, **kwargs) -> Dict[str, Any]:
        """Fetch URL with retry logic"""
        for attempt in range(max_retries):
            try:
                async with aiohttp.ClientSession() as session:
                    async with session.get(url, **kwargs) as response:
                        if response.status == 200:
                            return await response.json()
                        elif response.status >= 500 and attempt < max_retries - 1:
                            await asyncio.sleep(2 ** attempt)  # Exponential backoff
                            continue
                        else:
                            return {"error": f"HTTP {response.status}", "url": url}
            except Exception as e:
                if attempt < max_retries - 1:
                    await asyncio.sleep(2 ** attempt)
                    continue
                return {"error": str(e), "url": url}

# Usage example
api_client = ConcurrentAPIClient(max_concurrent=5)

# Fetch multiple URLs
urls = [
    "https://api.example.com/data1",
    "https://api.example.com/data2",
    "https://api.example.com/data3",
]

results = await api_client.fetch_multiple(urls)
for result in results:
    if "error" not in result:
        print(f"Success: {result}")
    else:
        print(f"Error: {result['error']}")

Response Standards

Consistent response formats with proper HTTP status codes and error handling.

# Standard response format
from typing import Generic, TypeVar, Optional
from pydantic import BaseModel
from fastapi import HTTPException

T = TypeVar('T')

class APIResponse(BaseModel, Generic[T]):
    """Standard API response format"""
    success: bool
    data: Optional[T] = None
    message: Optional[str] = None
    errors: Optional[list] = None
    metadata: Optional[dict] = None

class ErrorResponse(BaseModel):
    """Error response format"""
    success: bool = False
    error: str
    details: Optional[dict] = None
    request_id: Optional[str] = None

# Usage in routers
@router.get("/apps/{app_id}")
async def get_app(
    app_id: int,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user)
) -> APIResponse[AppResponse]:
    """Get application by ID"""
    try:
        app = await db.get(App, app_id)
        if not app or app.owner_id != current_user.id:
            raise HTTPException(
                status_code=404,
                detail="Application not found"
            )
        
        return APIResponse(
            success=True,
            data=AppResponse.from_orm(app),
            message="Application retrieved successfully"
        )
    except HTTPException:
        raise
    except Exception as e:
        return ErrorResponse(
            error="Internal server error",
            details={"error": str(e)}
        )

# Error handlers
@app.exception_handler(HTTPException)
async def http_exception_handler(request, exc):
    return JSONResponse(
        status_code=exc.status_code,
        content=ErrorResponse(
            error=exc.detail,
            request_id=request.headers.get("X-Request-ID")
        ).dict()
    )

    return JSONResponse(

        content=ErrorResponse(
            error="Internal server error",
            request_id=request.headers.get("X-Request-ID")
        ).dict()
    )
