Frontend Architecture: This document provides a comprehensive overview of WebCraft's frontend architecture, built with Next.js 15, React 18, and modern web technologies.

API Routes

API routes for server-side functionality with proper error handling and validation.

// app/api/auth/login/route.ts - API Route
import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'

const loginSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
})

export async function POST(request: NextRequest) {
  try {
    const body = await request.json()
    
    // Validate input
    const validatedData = loginSchema.parse(body)
    
    // Call backend API
    const response = await fetch(`${process.env.API_URL}/auth/login`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(validatedData),
    })
    
    if (!response.ok) {
      const error = await response.json()
      return NextResponse.json(
        { error: error.message || 'Login failed' },
        { status: response.status }
      )
    }
    
    const data = await response.json()
    
    // Set HTTP-only cookie
    const responseWithCookie = NextResponse.json(data)
    responseWithCookie.cookies.set('auth-token', data.access_token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'strict',
      maxAge: 60 * 60 * 24 * 7, // 7 days
    })
    
    return responseWithCookie
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Invalid input', details: error.errors },
        { status: 400 }
      )
    }
    
    console.error('Login error:', error)
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    )
  }
}

Custom Hooks

Reusable custom hooks for common functionality with proper TypeScript typing.

// hooks/useAuth.ts - Authentication hook
import { useState, useEffect, useCallback } from 'react'
import { useRouter } from 'next/navigation'
import { useAppDispatch, useAppSelector } from '@/store/hooks'
import { login, logout, selectAuth } from '@/store/slices/authSlice'

interface UseAuthReturn {
  user: User | null
  isAuthenticated: boolean
  isLoading: boolean
  error: string | null
  login: (email: string, password: string) => Promise
  logout: () => void
  clearError: () => void
}

export function useAuth(): UseAuthReturn {
  const dispatch = useAppDispatch()
  const router = useRouter()
  const { user, isLoading, error } = useAppSelector(selectAuth)
  
  const handleLogin = useCallback(async (email: string, password: string) => {
    try {
      await dispatch(login({ email, password })).unwrap()
      router.push('/dashboard')
    } catch (error) {
      // Error is handled in the slice
      console.error('Login failed:', error)
    }
  }, [dispatch, router])
  
  const handleLogout = useCallback(() => {
    dispatch(logout())
    router.push('/login')
  }, [dispatch, router])
  
  const clearError = useCallback(() => {
    dispatch(clearAuthError())
  }, [dispatch])
  
  return {
    user,
    isAuthenticated: !!user,
    isLoading,
    error,
    login: handleLogin,
    logout: handleLogout,
    clearError,
  }
}

// hooks/useEditor.ts - Editor state hook
import { useState, useCallback, useMemo } from 'react'
import { useAppDispatch, useAppSelector } from '@/store/hooks'
import { 
  selectEditor,
  addWidget,
  updateWidget,
  deleteWidget,
  setSelectedWidget,
} from '@/store/slices/editorSlice'

interface UseEditorReturn {
  widgets: Widget[]
  selectedWidget: Widget | null
  addWidget: (type: WidgetType, position: Position) => void
  updateWidget: (id: string, updates: Partial) => void
  deleteWidget: (id: string) => void
  selectWidget: (id: string | null) => void
  canUndo: boolean
  canRedo: boolean
}

export function useEditor(): UseEditorReturn {
  const dispatch = useAppDispatch()
  const { widgets, selectedWidget, history } = useAppSelector(selectEditor)
  
  const canUndo = useMemo(() => history.past.length > 0, [history.past])
  const canRedo = useMemo(() => history.future.length > 0, [history.future])
  
  const handleAddWidget = useCallback((type: WidgetType, position: Position) => {
    const newWidget: Widget = {
      id: generateId(),
      type,
      position,
      properties: getDefaultProperties(type),
      children: [],
    }
    dispatch(addWidget(newWidget))
  }, [dispatch])
  
  const handleUpdateWidget = useCallback((id: string, updates: Partial) => {
    dispatch(updateWidget({ id, updates }))
  }, [dispatch])
  
  const handleDeleteWidget = useCallback((id: string) => {
    dispatch(deleteWidget(id))
  }, [dispatch])
  
  const handleSelectWidget = useCallback((id: string | null) => {
    dispatch(setSelectedWidget(id))
  }, [dispatch])
  
  return {
    widgets,
    selectedWidget,
    addWidget: handleAddWidget,
    updateWidget: handleUpdateWidget,
    deleteWidget: handleDeleteWidget,
    selectWidget: handleSelectWidget,
    canUndo,
    canRedo,
  }

Slice Patterns

Well-structured Redux slices with async thunks and proper TypeScript typing.

// store/slices/editorSlice.ts - Editor state management
import { createSlice, createAsyncThunk, PayloadAction } from '@reduxjs/toolkit'
import { Widget, Position, WidgetType } from '@/types'
import { api } from '@/lib/api'

interface EditorState {
  widgets: Widget[]
  selectedWidget: string | null
  clipboard: Widget | null
  history: {
    past: Widget[][]
    present: Widget[]
    future: Widget[][]
  }
  isLoading: boolean
  error: string | null
}

const initialState: EditorState = {
  widgets: [],
  selectedWidget: null,
  clipboard: null,
  history: {
    past: [],
    present: [],
    future: [],
  },
  isLoading: false,
  error: null,
}

// Async thunks
export const loadAppWidgets = createAsyncThunk(
  'editor/loadAppWidgets',
  async (appId: string) => {
    const response = await api.get(`/apps/${appId}/widgets`)
    return response.data
  }
)

export const saveAppWidgets = createAsyncThunk(
  'editor/saveAppWidgets',
  async ({ appId, widgets }: { appId: string; widgets: Widget[] }) => {
    const response = await api.put(`/apps/${appId}/widgets`, { widgets })
    return response.data
  }
)

const editorSlice = createSlice({
  name: 'editor',
  initialState,
  reducers: {
    addWidget: (state, action: PayloadAction) => {
      state.widgets.push(action.payload)
      state.selectedWidget = action.payload.id
      saveToHistory(state)
    },
    
    updateWidget: (state, action: PayloadAction<{ id: string; updates: Partial }>) => {
      const { id, updates } = action.payload
      const widget = state.widgets.find(w => w.id === id)
      if (widget) {
        Object.assign(widget, updates)
        saveToHistory(state)
      }
    },
    
    deleteWidget: (state, action: PayloadAction) => {
      state.widgets = state.widgets.filter(w => w.id !== action.payload)
      if (state.selectedWidget === action.payload) {
        state.selectedWidget = null
      }
      saveToHistory(state)
    },
    
    setSelectedWidget: (state, action: PayloadAction) => {
      state.selectedWidget = action.payload
    },
    
    copyWidget: (state, action: PayloadAction) => {
      const widget = state.widgets.find(w => w.id === action.payload)
      if (widget) {
        state.clipboard = { ...widget, id: generateId() }
      }
    },
    
    pasteWidget: (state) => {
      if (state.clipboard) {
        state.widgets.push(state.clipboard)
        state.selectedWidget = state.clipboard.id
        saveToHistory(state)
      }
    },
    
    undo: (state) => {
      if (state.history.past.length > 0) {
        const previous = state.history.past[state.history.past.length - 1]
        state.history = {
          past: state.history.past.slice(0, -1),
          present: previous,
          future: [state.history.present, ...state.history.future],
        }
        state.widgets = previous
      }
    },
    
    redo: (state) => {
      if (state.history.future.length > 0) {
        const next = state.history.future[0]
        state.history = {
          past: [...state.history.past, state.history.present],
          present: next,
          future: state.history.future.slice(1),
        }
        state.widgets = next
      }
    },
  },
  extraReducers: (builder) => {
    builder
      .addCase(loadAppWidgets.pending, (state) => {
        state.isLoading = true
        state.error = null
      })
      .addCase(loadAppWidgets.fulfilled, (state, action) => {
        state.isLoading = false
        state.widgets = action.payload
        state.history.present = action.payload
      })
      .addCase(loadAppWidgets.rejected, (state, action) => {
        state.isLoading = false
        state.error = action.error.message || 'Failed to load widgets'
      })
      .addCase(saveAppWidgets.pending, (state) => {
        state.isLoading = true
        state.error = null
      })
      .addCase(saveAppWidgets.fulfilled, (state) => {
        state.isLoading = false
      })
      .addCase(saveAppWidgets.rejected, (state, action) => {
        state.isLoading = false
        state.error = action.error.message || 'Failed to save widgets'
      })
  },
})

function saveToHistory(state: EditorState) {
  state.history.past.push(state.history.present)
  state.history.present = [...state.widgets]
  state.history.future = []
}

export const {
  addWidget,
  updateWidget,
  deleteWidget,
  setSelectedWidget,
  copyWidget,
  pasteWidget,
  undo,
  redo,
} = editorSlice.actions

export default editorSlice.reducer

// components/editor/DroppableCanvas.tsx - Droppable canvas
import React from 'react'
import { useDrop } from 'react-dnd'
import { useAppDispatch } from '@/store/hooks'
import { addWidget } from '@/store/slices/editorSlice'

interface DroppableCanvasProps {
  children: React.ReactNode
  className?: string
}

export function DroppableCanvas({ children, className }: DroppableCanvasProps) {
  const dispatch = useAppDispatch()
  
  const [{ isOver }, drop] = useDrop({
    accept: ['widget', 'new-widget'],
    drop: (item: any, monitor) => {
      const offset = monitor.getClientOffset()
      if (offset) {
        const canvasRect = document.getElementById('canvas')?.getBoundingClientRect()
        if (canvasRect) {
          const position = {
            x: offset.x - canvasRect.left,
            y: offset.y - canvasRect.top,
          }
          
          if (item.type === 'new-widget') {
            // Create new widget
            const newWidget = {
              id: generateId(),
              type: item.widgetType,
              position,
              properties: getDefaultProperties(item.widgetType),
            }
            dispatch(addWidget(newWidget))
          } else {
            // Move existing widget
            dispatch(updateWidget({
              id: item.widget.id,
              updates: { position }
            }))
          }
        }
      }
    },
    collect: (monitor) => ({
      isOver: monitor.isOver(),
    }),
  })
  
  return (
    

      {children}
    

  )
}
