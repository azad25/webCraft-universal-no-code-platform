API Routers Guide

Creating and organizing API endpoints in WebCraft v2

Router Architecture

WebCraft v2 uses a domain-driven architecture with 34 specialized routers. Each domain has its own router, service, and schemas.

Domain Structure

├── auth/                    # Authentication

Step 2: Define Schemas (schemas.py)

"""My Domain schemas"""
from pydantic import BaseModel, Field
from typing import Optional, List, Dict, Any
from datetime import datetime
from enum import Enum


class ItemStatus(str, Enum):
    DRAFT = "draft"
    ACTIVE = "active"
    ARCHIVED = "archived"


class ItemCreate(BaseModel):
    """Schema for creating an item"""
    name: str = Field(..., min_length=1, max_length=255)
    description: Optional[str] = None
    status: ItemStatus = ItemStatus.DRAFT
    config: Dict[str, Any] = Field(default_factory=dict)
    tags: List[str] = Field(default_factory=list)
    
    class Config:
        json_schema_extra = {
            "example": {
                "name": "My Item",
                "description": "A sample item",
                "status": "draft",
                "config": {"key": "value"},
                "tags": ["tag1", "tag2"]
            }
        }


class ItemUpdate(BaseModel):
    """Schema for updating an item"""
    name: Optional[str] = Field(None, min_length=1, max_length=255)
    description: Optional[str] = None
    status: Optional[ItemStatus] = None
    config: Optional[Dict[str, Any]] = None
    tags: Optional[List[str]] = None


class ItemResponse(BaseModel):
    """Schema for item response"""
    id: str
    app_id: str
    name: str
    description: Optional[str]
    status: str
    config: Dict[str, Any]
    tags: List[str]
    created_at: datetime
    updated_at: Optional[datetime]
    
    class Config:
        from_attributes = True


class ItemListResponse(BaseModel):
    """Schema for paginated item list"""
    items: List[ItemResponse]
    total: int
    page: int
    page_size: int
    has_more: bool

Step 3: Create Service (service.py)

"""My Domain service"""
from typing import Optional, Dict, Any, List
from sqlalchemy.orm import Session
from datetime import datetime
import uuid

# In-memory storage (replace with database in production)
items_store: Dict[str, Dict] = {}


class MyDomainService:
    """Service class for my domain business logic"""
    
    def __init__(self, db: Session):
        self.db = db
    
    async def create_item(
        self, 
        app_id: str, 
        item_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """Create a new item"""
        item_id = str(uuid.uuid4())
        
        item = {
            "id": item_id,
            "app_id": app_id,
            "name": item_data["name"],
            "description": item_data.get("description"),
            "status": item_data.get("status", "draft"),
            "config": item_data.get("config", {}),
            "tags": item_data.get("tags", []),
            "created_at": datetime.utcnow().isoformat(),
            "updated_at": None
        }
        
        items_store[item_id] = item
        return item
    
    async def list_items(
        self,
        app_id: str,
        status: Optional[str] = None,
        search: Optional[str] = None,
        page: int = 1,
        page_size: int = 20
    ) -> Dict[str, Any]:
        """List items with filtering and pagination"""
        # Filter by app_id
        items = [i for i in items_store.values() if i["app_id"] == app_id]
        
        # Apply filters
        if status:
            items = [i for i in items if i["status"] == status]
        
        if search:
            search_lower = search.lower()
            items = [i for i in items if search_lower in i["name"].lower()]
        
        # Sort by created_at descending
        items.sort(key=lambda x: x["created_at"], reverse=True)
        
        # Paginate
        total = len(items)
        start = (page - 1) * page_size
        end = start + page_size
        paginated_items = items[start:end]
        
        return {
            "items": paginated_items,
            "total": total,
            "page": page,
            "page_size": page_size,
            "has_more": end < total
        }
    
    async def get_item(
        self, 
        app_id: str, 
        item_id: str
    ) -> Optional[Dict[str, Any]]:
        """Get a single item by ID"""
        item = items_store.get(item_id)
        if item and item["app_id"] == app_id:
            return item
        return None
    
    async def update_item(
        self,
        app_id: str,
        item_id: str,
        updates: Dict[str, Any]
    ) -> Optional[Dict[str, Any]]:
        """Update an existing item"""
        item = items_store.get(item_id)
        if not item or item["app_id"] != app_id:
            return None
        
        # Apply updates
        for key, value in updates.items():
            if value is not None:
                item[key] = value
        
        item["updated_at"] = datetime.utcnow().isoformat()
        return item
    
    async def delete_item(
        self, 
        app_id: str, 
        item_id: str
    ) -> bool:
        """Delete an item"""
        item = items_store.get(item_id)
        if item and item["app_id"] == app_id:
            del items_store[item_id]
            return True
        return False
    
    async def bulk_update(
        self,
        app_id: str,
        item_ids: List[str],
        updates: Dict[str, Any]
    ) -> Dict[str, Any]:
        """Bulk update multiple items"""
        updated = 0
        failed = 0
        
        for item_id in item_ids:
            result = await self.update_item(app_id, item_id, updates)
            if result:
                updated += 1
            else:
                failed += 1
        
        return {"updated": updated, "failed": failed}
    
    async def get_stats(self, app_id: str) -> Dict[str, Any]:
        """Get statistics for items"""
        items = [i for i in items_store.values() if i["app_id"] == app_id]
        
        return {
            "total": len(items),
            "by_status": {
                "draft": len([i for i in items if i["status"] == "draft"]),
                "active": len([i for i in items if i["status"] == "active"]),
                "archived": len([i for i in items if i["status"] == "archived"])
            }
        }

Step 4: Create Router (router.py)

"""My Domain router"""
from fastapi import APIRouter, Depends, HTTPException, Query, Path, status
from sqlalchemy.orm import Session
from typing import Optional, List
import uuid

from src.core.database import get_db
from src.core.security import get_current_user
from src.domains.auth.schemas import UserResponse
from src.domains.apps.models import App
from .service import MyDomainService
from .schemas import (
    ItemCreate, 
    ItemUpdate, 
    ItemResponse, 
    ItemListResponse
)

router = APIRouter(prefix="/apps/{app_id}/items")


# Helper function to verify app ownership
async def get_app_or_404(
    app_id: uuid.UUID, 
    user_id: str, 
    db: Session
) -> App:
    """Get app and verify ownership"""
    app = db.query(App).filter(
        App.id == app_id, 
        App.owner_id == user_id
    ).first()
    if not app:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail="App not found"
        )
    return app


@router.post("", response_model=ItemResponse, status_code=status.HTTP_201_CREATED)
async def create_item(
    app_id: uuid.UUID = Path(..., description="App ID"),
    item: ItemCreate = None,
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """
    Create a new item.
    
    - **name**: Item name (required)
    - **description**: Optional description
    - **status**: draft, active, or archived
    - **config**: Custom configuration object
    - **tags**: List of tags
    """
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    result = await service.create_item(str(app_id), item.model_dump())
    return result


@router.get("", response_model=ItemListResponse)
async def list_items(
    app_id: uuid.UUID = Path(..., description="App ID"),
    status: Optional[str] = Query(None, description="Filter by status"),
    search: Optional[str] = Query(None, description="Search by name"),
    page: int = Query(1, ge=1, description="Page number"),
    page_size: int = Query(20, ge=1, le=100, description="Items per page"),
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """
    List all items for an app with filtering and pagination.
    
    - **status**: Filter by item status
    - **search**: Search items by name
    - **page**: Page number (default: 1)
    - **page_size**: Items per page (default: 20, max: 100)
    """
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    return await service.list_items(
        str(app_id), 
        status=status, 
        search=search,
        page=page, 
        page_size=page_size
    )


@router.get("/{item_id}", response_model=ItemResponse)
async def get_item(
    app_id: uuid.UUID = Path(..., description="App ID"),
    item_id: str = Path(..., description="Item ID"),
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Get a single item by ID."""
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    item = await service.get_item(str(app_id), item_id)
    
    if not item:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail="Item not found"
        )
    return item


@router.put("/{item_id}", response_model=ItemResponse)
async def update_item(
    app_id: uuid.UUID = Path(..., description="App ID"),
    item_id: str = Path(..., description="Item ID"),
    updates: ItemUpdate = None,
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Update an existing item."""
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    item = await service.update_item(
        str(app_id), 
        item_id, 
        updates.model_dump(exclude_unset=True)
    )
    
    if not item:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail="Item not found"
        )
    return item


@router.delete("/{item_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_item(
    app_id: uuid.UUID = Path(..., description="App ID"),
    item_id: str = Path(..., description="Item ID"),
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Delete an item."""
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    if not await service.delete_item(str(app_id), item_id):
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, 
            detail="Item not found"
        )


@router.post("/bulk-update")
async def bulk_update_items(
    app_id: uuid.UUID = Path(..., description="App ID"),
    item_ids: List[str] = None,
    updates: ItemUpdate = None,
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Bulk update multiple items."""
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    return await service.bulk_update(
        str(app_id), 
        item_ids, 
        updates.model_dump(exclude_unset=True)
    )


@router.get("/stats")
async def get_item_stats(
    app_id: uuid.UUID = Path(..., description="App ID"),
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Get item statistics for an app."""
    await get_app_or_404(app_id, str(current_user.id), db)
    
    service = MyDomainService(db)
    return await service.get_stats(str(app_id))

File Uploads

from fastapi import File, UploadFile

@router.post("/upload")
async def upload_file(
    app_id: uuid.UUID = Path(...),
    file: UploadFile = File(...),
    current_user: UserResponse = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Upload a file"""
    # Validate file type
    allowed_types = ["image/jpeg", "image/png", "image/gif"]
    if file.content_type not in allowed_types:
        raise HTTPException(400, "Invalid file type")
    
    # Process file
    contents = await file.read()
    # Save to storage...
    
    return {"filename": file.filename, "size": len(contents)}

WebSocket Endpoints

from fastapi import WebSocket, WebSocketDisconnect

@router.websocket("/ws/{app_id}")
async def websocket_endpoint(
    websocket: WebSocket,
    app_id: str
):
    """WebSocket connection for real-time updates"""
    await websocket.accept()
    
    try:
        while True:
            data = await websocket.receive_json()
            # Process message
            await websocket.send_json({"status": "received"})
    except WebSocketDisconnect:
        # Handle disconnect
        pass

Error Handling

from fastapi import HTTPException, status
from src.common.exceptions import AppException

# Using HTTPException
@router.get("/{item_id}")
async def get_item(item_id: str):
    item = await service.get_item(item_id)
    if not item:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Item not found"
        )
    return item

# Using custom AppException
class ItemNotFoundError(AppException):
    def __init__(self, item_id: str):
        super().__init__(
            status_code=404,
            error_code="ITEM_NOT_FOUND",
            message=f"Item {item_id} not found"
        )

@router.get("/{item_id}")
async def get_item(item_id: str):
    item = await service.get_item(item_id)
    if not item:
        raise ItemNotFoundError(item_id)
    return item

Response Models

from fastapi.responses import JSONResponse, StreamingResponse
import io

# Custom JSON response
@router.get("/export")
async def export_data(app_id: uuid.UUID):
    data = await service.get_all_data(str(app_id))
    return JSONResponse(
        content=data,
        headers={"X-Custom-Header": "value"}
    )

# Streaming response for large files
@router.get("/download")
async def download_file(app_id: uuid.UUID, file_id: str):
    file_content = await service.get_file(file_id)
    
    return StreamingResponse(
        io.BytesIO(file_content),
        media_type="application/octet-stream",
        headers={
            "Content-Disposition": f"attachment; filename={file_id}.zip"
        }
    )
