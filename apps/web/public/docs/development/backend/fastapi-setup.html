FastAPI Setup

Complete guide to setting up FastAPI for WebCraft backend development

Overview

WebCraft uses FastAPI as its primary backend framework, providing high performance, automatic API documentation, and type safety. This guide covers the complete setup process, configuration, and best practices.

Key Features:
  • High performance with async support
  • Automatic API documentation (Swagger/OpenAPI)
  • Type hints and validation with Pydantic
  • Built-in dependency injection
  • WebSocket support for real-time features

Project Structure

The backend follows a modular architecture with clear separation of concerns:

apps/api/
├── main.py                 # Application entry point
├── core/                   # Core functionality
│   ├── config.py          # Configuration management
│   ├── database.py        # Database connection and models

├── routers/               # API route definitions

│   ├── ai_service.py     # AI integration
│   ├── deployment_service.py
│   └── ...
├── requirements.txt      # Dependencies
└── alembic/             # Database migrations

Configuration Class

# core/config.py
from typing import List
from pydantic_settings import BaseSettings
from pydantic import AnyHttpUrl, validator


class Settings(BaseSettings):
    PROJECT_NAME: str = "WebCraft API"
    VERSION: str = "1.0.0"
    DEBUG: bool = False
    
    # API
    API_V1_STR: str = "/api/v1"
    
    # Security
    SECRET_KEY: str
    ALGORITHM: str = "HS256"
    ACCESS_TOKEN_EXPIRE_MINUTES: int = 30
    
    # Database
    DATABASE_URL: str
    
    # Redis
    REDIS_URL: str = "redis://localhost:6379/0"
    
    # CORS
    BACKEND_CORS_ORIGINS: List[AnyHttpUrl] = []
    
    @validator("BACKEND_CORS_ORIGINS", pre=True)
    def assemble_cors_origins(cls, v):
        if isinstance(v, str) and not v.startswith("["):
            return [i.strip() for i in v.split(",")]
        return v
    
    class Config:
        env_file = ".env"
        case_sensitive = True


settings = Settings()

Main Application

Application Factory Pattern

# main.py
from contextlib import asynccontextmanager
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from fastapi.middleware.gzip import GZipMiddleware

from core.config import settings
from core.database import engine, Base
from routers import auth, apps, widgets, pages, automations
from core.module_system import ModuleSystem


@asynccontextmanager
async def lifespan(app: FastAPI):
    """Handle application lifecycle events."""
    # Startup
    Base.metadata.create_all(bind=engine)
    
    # Initialize module system
    module_system = ModuleSystem()
    await module_system.initialize()
    
    yield
    
    # Shutdown
    # Cleanup resources


def create_application() -> FastAPI:
    """Create and configure the FastAPI application."""
    app = FastAPI(
        title=settings.PROJECT_NAME,
        version=settings.VERSION,
        openapi_url=f"{settings.API_V1_STR}/openapi.json",
        docs_url=f"{settings.API_V1_STR}/docs",
        redoc_url=f"{settings.API_V1_STR}/redoc",
        debug=settings.DEBUG,
        lifespan=lifespan,
    )
    
    # Add middleware
    app.add_middleware(GZipMiddleware, minimum_size=1000)
    
    if settings.BACKEND_CORS_ORIGINS:
        app.add_middleware(
            CORSMiddleware,
            allow_origins=[str(origin) for origin in settings.BACKEND_CORS_ORIGINS],
            allow_credentials=True,
            allow_methods=["*"],
            allow_headers=["*"],
        )
    
    # Include routers
    app.include_router(auth.router, prefix=settings.API_V1_STR)
    app.include_router(apps.router, prefix=f"{settings.API_V1_STR}/apps", tags=["apps"])
    app.include_router(widgets.router, prefix=f"{settings.API_V1_STR}/widgets", tags=["widgets"])
    app.include_router(pages.router, prefix=f"{settings.API_V1_STR}/pages", tags=["pages"])
    app.include_router(automations.router, prefix=f"{settings.API_V1_STR}/automations", tags=["automations"])
    
    return app


app = create_application()


@app.get("/health")
async def health_check():
    """Health check endpoint."""
    return {"status": "healthy", "version": settings.VERSION}

Database Setup

Database Configuration

# core/database.py
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker

from core.config import settings

engine = create_engine(
    settings.DATABASE_URL,
    pool_pre_ping=True,
    pool_recycle=300,
    pool_size=10,
    max_overflow=20,
)

SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

Base = declarative_base()


def get_db():
    """Database session dependency."""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

Router Structure

Example Router Implementation

# routers/apps.py
from typing import List
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session

from core.database import get_db
from core.auth import get_current_user
from core.database import User, App
from services.app_builder import AppBuilderService

router = APIRouter()


@router.get("/", response_model=List[dict])
async def get_apps(
    skip: int = 0,
    limit: int = 100,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """Get all apps for the current user."""
    apps = db.query(App).filter(
        App.owner_id == current_user.id,
        App.is_active == True
    ).offset(skip).limit(limit).all()
    
    return [{"id": str(app.id), "name": app.name, "slug": app.slug} for app in apps]


@router.post("/", status_code=status.HTTP_201_CREATED)
async def create_app(
    app_data: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """Create a new app."""
    service = AppBuilderService(db)
    app = await service.create_app(
        name=app_data["name"],
        app_type=app_data.get("app_type", "website"),
        owner_id=current_user.id
    )
    return {"id": str(app.id), "name": app.name, "slug": app.slug}

Services Layer

WebCraft uses a service layer pattern to separate business logic from API routes. Key services include:

App Builder Service

# services/app_builder.py
from sqlalchemy.orm import Session
from core.database import App, Page
import uuid

class AppBuilderService:
    def __init__(self, db: Session):
        self.db = db
    
    async def create_app(self, name: str, app_type: str, owner_id: str):
        """Create a new app with default pages."""
        app = App(
            id=uuid.uuid4(),
            name=name,
            slug=self._generate_slug(name),
            app_type=app_type,
            owner_id=owner_id
        )
        
        self.db.add(app)
        self.db.commit()
        self.db.refresh(app)
        
        # Create default homepage
        homepage = Page(
            id=uuid.uuid4(),
            app_id=app.id,
            title="Home",
            slug="home",
            is_homepage=True,
            content={"elements": []}
        )
        self.db.add(homepage)
        self.db.commit()
        
        return app
    
    def _generate_slug(self, name: str) -> str:
        """Generate URL-friendly slug from name."""
        return name.lower().replace(" ", "-")

Best Practices

API Design

  • Use RESTful conventions for resource naming
  • Implement proper HTTP status codes
  • Version your APIs (e.g., /api/v1/)
  • Use consistent response formats
  • Implement proper error handling
  • Document all endpoints with descriptions

Database Operations

  • Use SQLAlchemy ORM for database operations
  • Implement proper database migrations with Alembic
  • Use connection pooling for better performance
  • Implement proper indexing for frequently queried columns
  • Use database transactions appropriately

Security

  • Implement JWT authentication
  • Use HTTPS in production
  • Validate all input data with Pydantic
  • Implement rate limiting
  • Use environment variables for sensitive configuration
  • Implement proper CORS policies
