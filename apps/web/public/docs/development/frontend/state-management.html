State Management

Understanding Redux Toolkit and state management in WebCraft

Overview

WebCraft uses Redux Toolkit for global state management, combined with React Context for component-level state. This provides a scalable and maintainable state management solution.

State Architecture

The state is organized into slices, each managing a specific domain:

  • authSlice - Authentication state
  • appSlice - App management state
  • editorSlice - Visual editor state
  • uiSlice - UI state (modals, sidebars, etc.)
  • moduleSlice - Module system state
  • collaborationSlice - Real-time collaboration state

Redux Store Setup

The Redux store is configured in store/index.ts:

import { configureStore } from '@reduxjs/toolkit'
import { persistStore, persistReducer } from 'redux-persist'
import storage from 'redux-persist/lib/storage'

import authReducer from './slices/authSlice'
import appReducer from './slices/appSlice'
import editorReducer from './slices/editorSlice'
import uiReducer from './slices/uiSlice'

const persistConfig = {
  key: 'root',
  storage,
  whitelist: ['auth', 'app'] // Only persist auth and app state
}

const rootReducer = combineReducers({
  auth: authReducer,
  app: appReducer,
  editor: editorReducer,
  ui: uiReducer,
})

const persistedReducer = persistReducer(persistConfig, rootReducer)

export const store = configureStore({
  reducer: persistedReducer,
  middleware: (getDefaultMiddleware) =>
    getDefaultMiddleware({
      serializableCheck: {
        ignoredActions: [FLUSH, REHYDRATE, PAUSE, PERSIST, PURGE, REGISTER],
      },
    }),
})

export const persistor = persistStore(store)
export type RootState = ReturnType
export type AppDispatch = typeof store.dispatch

Creating a Slice

Here's how to create a Redux slice:

import { createSlice, PayloadAction } from '@reduxjs/toolkit'

interface MyState {
  items: string[]
  loading: boolean
  error: string | null
}

const initialState: MyState = {
  items: [],
  loading: false,
  error: null
}

const mySlice = createSlice({
  name: 'myFeature',
  initialState,
  reducers: {
    setLoading: (state, action: PayloadAction) => {
      state.loading = action.payload
    },
    addItem: (state, action: PayloadAction) => {
      state.items.push(action.payload)
    },
    removeItem: (state, action: PayloadAction) => {
      state.items = state.items.filter(item => item !== action.payload)
    },
    setError: (state, action: PayloadAction) => {
      state.error = action.payload
    }
  }
})

export const { setLoading, addItem, removeItem, setError } = mySlice.actions
export default mySlice.reducer

Editor Slice

The editor slice manages visual editor state:

interface EditorState {
  // Current app
  appId: string | null
  appName: string
  
  // Pages
  pages: Record
  currentPageId: string | null
  
  // Elements
  elements: Record
  
  // Selection
  selectedElementId: string | null
  hoveredElementId: string | null
  multiSelectedIds: string[]
  
  // History (undo/redo)
  history: HistoryEntry[]
  historyIndex: number
  
  // Canvas
  zoom: number
  panOffset: Position
  gridEnabled: boolean
  
  // Saving
  isDirty: boolean
  isSaving: boolean
}

const editorSlice = createSlice({
  name: 'editor',
  initialState,
  reducers: {
    setApp: (state, action: PayloadAction<{ id: string; name: string }>) => {
      state.appId = action.payload.id
      state.appName = action.payload.name
    },
    
    addElement: (state, action: PayloadAction) => {
      const element = action.payload
      state.elements[element.id] = element
      saveToHistory(state)
    },
    
    updateElement: (state, action: PayloadAction<{ id: string; updates: Partial }>) => {
      const { id, updates } = action.payload
      if (state.elements[id]) {
        state.elements[id] = { ...state.elements[id], ...updates }
        saveToHistory(state)
      }
    },
    
    deleteElement: (state, action: PayloadAction) => {
      delete state.elements[action.payload]
      saveToHistory(state)
    },
    
    selectElement: (state, action: PayloadAction) => {
      state.selectedElementId = action.payload
    },
    
    undo: (state) => {
      if (state.historyIndex > 0) {
        state.historyIndex--
        state.elements = JSON.parse(JSON.stringify(state.history[state.historyIndex].elements))
      }
    },
    
    redo: (state) => {
      if (state.historyIndex < state.history.length - 1) {
        state.historyIndex++
        state.elements = JSON.parse(JSON.stringify(state.history[state.historyIndex].elements))
      }
    }
  }
})

RTK Query for API State

RTK Query handles server state and API calls:

import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react'

export const apiSlice = createApi({
  reducerPath: 'api',
  baseQuery: fetchBaseQuery({
    baseUrl: '/api',
    prepareHeaders: (headers, { getState }) => {
      const token = (getState() as RootState).auth.token
      if (token) {
        headers.set('authorization', `Bearer ${token}`)
      }
      return headers
    },
  }),
  tagTypes: ['App', 'Page', 'Widget'],
  endpoints: (builder) => ({
    getApps: builder.query({
      query: () => '/apps',
      providesTags: ['App'],
    }),
    
    getApp: builder.query({
      query: (id) => `/apps/${id}`,
      providesTags: (result, error, id) => [{ type: 'App', id }],
    }),
    
    createApp: builder.mutation>({
      query: (body) => ({
        url: '/apps',
        method: 'POST',
        body,
      }),
      invalidatesTags: ['App'],
    }),
    
    updateApp: builder.mutation }>({
      query: ({ id, data }) => ({
        url: `/apps/${id}`,
        method: 'PUT',
        body: data,
      }),
      invalidatesTags: (result, error, { id }) => [{ type: 'App', id }],
    }),
  }),
})

export const {
  useGetAppsQuery,
  useGetAppQuery,
  useCreateAppMutation,
  useUpdateAppMutation,
} = apiSlice

React Context for Component State

Use React Context for component-specific state that doesn't need to be global:

import { createContext, useContext, useState } from 'react'

interface EditorContextValue {
  selectedElement: Element | null
  selectElement: (element: Element | null) => void
  addElement: (element: Omit) => void
  updateElement: (id: string, updates: Partial) => void
}

const EditorContext = createContext(null)

export function EditorProvider({ children }: { children: React.ReactNode }) {
  const [selectedElement, setSelectedElement] = useState(null)
  const [elements, setElements] = useState([])
  
  const selectElement = (element: Element | null) => {
    setSelectedElement(element)
  }
  
  const addElement = (element: Omit) => {
    const newElement = { ...element, id: uuidv4() }
    setElements(prev => [...prev, newElement])
  }
  
  const updateElement = (id: string, updates: Partial) => {
    setElements(prev =>
      prev.map(el => el.id === id ? { ...el, ...updates } : el)
    )
  }
  
  return (
    
      {children}
    
  )
}

export function useEditor() {
  const context = useContext(EditorContext)
  if (!context) {
    throw new Error('useEditor must be used within EditorProvider')
  }
  return context
}

State Persistence

Redux Persist automatically saves and restores state:

// Configure what to persist
const persistConfig = {
  key: 'root',
  storage,
  whitelist: ['auth', 'app'], // Only persist these slices
  blacklist: ['editor'], // Don't persist editor state
}

// State is automatically saved to localStorage
// and restored on app load

Best Practices

  • Use Redux for Global State - Auth, app data, editor state
  • Use Context for Component State - Local component state
  • Use RTK Query for Server State - API data, caching, synchronization
  • Keep Slices Focused - One slice per domain
  • Use TypeScript - Type safety for state and actions
  • Normalize State - Use records/objects for lookups
  • Avoid Deep Nesting - Keep state structure flat
