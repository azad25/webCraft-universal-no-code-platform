Editor System

Understanding the visual editor architecture and implementation

Overview

The WebCraft visual editor is a Figma-like drag-and-drop interface built with React, React DnD, and Framer Motion. It provides real-time editing, multi-device preview, and a comprehensive widget library.

Editor Architecture

The editor consists of several key components:

  • LiveEditor - Main editor container
  • EditorCanvas - Canvas where elements are rendered
  • EditorSidebar - Widget library and page management
  • PropertiesPanel - Element property editor
  • WidgetRenderer - Renders widget components
  • ResizeHandles - Element resize functionality
  • InlineEditor - Inline text editing

Saving Changes

The editor automatically saves changes to the backend:

async function savePage() {
  const { currentPageId, elements, appId } = useEditor()
  
  try {
    setIsSaving(true)
    
    await apiClient.put(`/api/apps/${appId}/pages/${currentPageId}`, {
      content: {
        elements: elements
      }
    })
    
    setIsDirty(false)
    console.log('✅ Page saved')
  } catch (error) {
    console.error('❌ Failed to save:', error)
  } finally {
    setIsSaving(false)
  }
}

// Auto-save on changes
useEffect(() => {
  if (isDirty) {
    const timer = setTimeout(() => {
      savePage()
    }, 2000) // Save after 2 seconds of inactivity
    
    return () => clearTimeout(timer)
  }
}, [isDirty])

Undo/Redo System

The editor maintains a history for undo/redo:

interface HistoryEntry {
  elements: Element[]
  timestamp: number
}

function saveToHistory(elements: Element[]) {
  const entry: HistoryEntry = {
    elements: JSON.parse(JSON.stringify(elements)),
    timestamp: Date.now()
  }
  
  // Remove future history if we're not at the end
  const newHistory = history.slice(0, historyIndex + 1)
  newHistory.push(entry)
  
  // Limit history length
  if (newHistory.length > maxHistoryLength) {
    newHistory.shift()
  }
  
  setHistory(newHistory)
  setHistoryIndex(newHistory.length - 1)
}

function undo() {
  if (historyIndex > 0) {
    setHistoryIndex(historyIndex - 1)
    setElements(history[historyIndex - 1].elements)
  }
}

function redo() {
  if (historyIndex < history.length - 1) {
    setHistoryIndex(historyIndex + 1)
    setElements(history[historyIndex + 1].elements)
  }
}
