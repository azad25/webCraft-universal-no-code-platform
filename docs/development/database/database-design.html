Database Design & Architecture

Comprehensive guide to WebCraft's database architecture and design patterns

Overview

WebCraft uses a multi-database architecture designed for scalability, performance, and reliability. Our database layer consists of:

  • PostgreSQL 15 - Primary relational database for structured data
  • Redis 7 - In-memory data store for caching and session management
  • TimescaleDB - Time-series data extension for analytics and metrics

Database Architecture

Multi-Tenant Architecture

WebCraft implements a multi-tenant architecture with row-level security:

-- Tenant isolation using PostgreSQL RLS
CREATE TABLE workspaces (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(255) UNIQUE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

ALTER TABLE workspaces ENABLE ROW LEVEL SECURITY;

CREATE POLICY workspace_isolation ON workspaces
    USING (id = current_setting('app.current_workspace_id')::UUID);

Performance Optimization

Indexing Strategy

-- Performance indexes
CREATE INDEX idx_workspaces_slug ON workspaces (slug);
CREATE INDEX idx_projects_workspace ON projects (workspace_id);
CREATE INDEX idx_pages_project_path ON pages (project_id, path);
CREATE INDEX idx_components_project_type ON components (project_id, type);

-- Composite indexes for complex queries
CREATE INDEX idx_page_analytics_workspace_time ON page_analytics (workspace_id, time DESC);
CREATE INDEX idx_workspace_members_user_role ON workspace_members (user_id, role);

-- Partial indexes for filtered queries
CREATE INDEX idx_pages_published ON pages (project_id) WHERE is_published = true;
CREATE INDEX idx_users_active ON auth.users (email) WHERE is_active = true;

Partitioning Strategy

-- Partition large tables by workspace
CREATE TABLE page_analytics_partitioned (
    LIKE page_analytics INCLUDING ALL
) PARTITION BY RANGE (workspace_id);

-- Create partitions for each workspace range
CREATE TABLE page_analytics_workspace_1 PARTITION OF page_analytics_partitioned
    FOR VALUES FROM ('00000000-0000-0000-0000-000000000000') TO ('10000000-0000-0000-0000-000000000000');

-- Function to automatically create partitions
CREATE OR REPLACE FUNCTION create_workspace_partition(workspace_id UUID)
RETURNS VOID AS $$
BEGIN
    EXECUTE format('
        CREATE TABLE IF NOT EXISTS page_analytics_workspace_%s 
        PARTITION OF page_analytics_partitioned 
        FOR VALUES FROM (%L) TO (%L)',
        replace(workspace_id::text, '-', '_'),
        workspace_id,
        workspace_id
    );
END;
$$ LANGUAGE plpgsql;

Redis Caching Architecture

Cache Structure

# Redis cache keys structure
CACHE_KEYS = {
    'workspace': 'workspace:{workspace_id}',
    'project': 'project:{project_id}',
    'page': 'page:{page_id}',
    'component': 'component:{component_id}',
    'user_session': 'session:{session_id}',
    'rate_limit': 'rate_limit:{user_id}:{endpoint}',
    'analytics': 'analytics:{workspace_id}:{date}',
}

# Cache TTL settings
CACHE_TTL = {
    'workspace': 3600,      # 1 hour
    'project': 1800,         # 30 minutes
    'page': 900,             # 15 minutes
    'component': 3600,       # 1 hour
    'user_session': 86400,   # 24 hours
    'rate_limit': 60,        # 1 minute
    'analytics': 300,        # 5 minutes
}

Caching Patterns

# Cache-aside pattern implementation
async def get_workspace_with_cache(workspace_id: str) -> Workspace:
    cache_key = CACHE_KEYS['workspace'].format(workspace_id=workspace_id)
    
    # Try cache first
    cached_data = await redis_client.get(cache_key)
    if cached_data:
        return Workspace.parse_raw(cached_data)
    
    # Cache miss - fetch from database
    workspace = await db.get_workspace(workspace_id)
    if workspace:
        # Store in cache with TTL
        await redis_client.setex(
            cache_key,
            CACHE_TTL['workspace'],
            workspace.json()
        )
    
    return workspace

# Cache invalidation
async def invalidate_workspace_cache(workspace_id: str):
    cache_key = CACHE_KEYS['workspace'].format(workspace_id=workspace_id)
    await redis_client.delete(cache_key)
    
    # Invalidate related caches
    project_keys = await redis_client.keys(f"project:*:workspace:{workspace_id}")
    if project_keys:
        await redis_client.delete(*project_keys)

Data Migration Strategy

# Alembic migration example
from alembic import op
import sqlalchemy as sa

def upgrade():
    # Create new table with proper constraints
    op.create_table(
        'new_table',
        sa.Column('id', sa.UUID(), nullable=False),
        sa.Column('name', sa.String(255), nullable=False),
        sa.Column('created_at', sa.DateTime(timezone=True), nullable=False),
        sa.PrimaryKeyConstraint('id'),
        sa.UniqueConstraint('name')
    )
    
    # Migrate data from old table
    op.execute("""
        INSERT INTO new_table (id, name, created_at)
        SELECT id, name, CURRENT_TIMESTAMP FROM old_table
    """)
    
    # Drop old table
    op.drop_table('old_table')

@app.get("/health/database")
async def database_health():
    try:
        # Test database connection
        async with db_session() as session:
            result = await session.execute(text("SELECT 1"))
            
        # Test Redis connection
        redis_ping = await redis_client.ping()
        
        return {
            "status": "healthy",
            "database": "connected",
            "redis": "connected" if redis_ping else "disconnected",
            "timestamp": datetime.utcnow().isoformat()
        }
    except Exception as e:
        return {
            "status": "unhealthy",
            "error": str(e),
            "timestamp": datetime.utcnow().isoformat()
        }
