Quick Code Example

Creating a Custom Widget (Backend)

from pydantic import BaseModel
from typing import Dict, Any, Optional
from core.module_system import BaseWidget

class CustomWidget(BaseWidget):
    class Config(BaseModel):
        title: str = "My Widget"
        content: str = ""
        style: Dict[str, Any] = {}
        
    def render(self, config: Config) -> Dict[str, Any]:
        return {
            "type": "custom-widget",
            "props": {
                "title": config.title,
                "content": config.content,
                "style": config.style
            }
        }
        
    def get_schema(self) -> Dict[str, Any]:
        return {
            "title": {"type": "string", "default": "My Widget"},
            "content": {"type": "richtext", "default": ""},
            "style": {"type": "object", "default": {}}
        }

Creating a Custom Widget (Frontend)

import React from 'react';
import { WidgetProps } from '@/types/widgets';

interface CustomWidgetProps extends WidgetProps {
  title: string;
  content: string;
  style?: React.CSSProperties;
}

export const CustomWidget: React.FC = ({
  title,
  content,
  style,
  onEdit,
  isSelected
}) => {
  return (
    <div 
      className={`custom-widget ${isSelected ? 'selected' : ''}`}
      style={style}
      onClick={onEdit}
    >
      <h3>{title}</h3>
      <div dangerouslySetInnerHTML={{ __html: content }} />
    </div>
  );
};

export default CustomWidget;
