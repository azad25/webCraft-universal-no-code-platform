Workspace Selector Component: A reusable dropdown component for workspace selection with loading states and error handling.
WorkspaceSelector.tsx 
import React, { useState, useEffect } from 'react';
import { useWorkspace } from '@/hooks/useWorkspace';
import { ChevronDownIcon } from '@heroicons/react/24/outline';

interface Workspace {
  id: string;
  name: string;
  slug: string;
  role: 'owner' | 'admin' | 'member';
}

interface WorkspaceSelectorProps {
  onWorkspaceChange?: (workspace: Workspace) => void;
  className?: string;
}

export const WorkspaceSelector: React.FC<WorkspaceSelectorProps> = ({
  onWorkspaceChange,
  className = ''
}) => {
  const { workspaces, currentWorkspace, loading, error, switchWorkspace } = useWorkspace();
  const [isOpen, setIsOpen] = useState(false);

  const handleWorkspaceSelect = async (workspace: Workspace) => {
    try {
      await switchWorkspace(workspace.id);
      onWorkspaceChange?.(workspace);
      setIsOpen(false);
    } catch (error) {
      console.error('Failed to switch workspace:', error);
    }
  };

  if (loading) {
    return (
      <div className={`animate-pulse bg-gray-200 rounded-md h-10 w-48 ${className}`}>
        <div className="h-4 bg-gray-300 rounded w-3/4 mx-auto mt-3"></div>
      </div>
    );
  }

  if (error) {
    return (
      <div className={`bg-red-50 border border-red-200 rounded-md p-3 text-red-700 ${className}`}>
        Failed to load workspaces
      </div>
    );
  }

  return (
    <div className={`relative ${className}`}>
      <button
        onClick={() => setIsOpen(!isOpen)}
        className="flex items-center justify-between w-full px-4 py-2 text-left bg-white border border-gray-300 rounded-md shadow-sm hover:bg-gray-50 focus:outline-none focus:ring-2 focus:ring-blue-500"
      >
        <span className="flex items-center">
          <div className="w-8 h-8 bg-blue-500 rounded-full flex items-center justify-center text-white text-sm font-medium mr-3">
            {currentWorkspace?.name?.charAt(0)?.toUpperCase()}
          </div>
          <span className="text-gray-900">{currentWorkspace?.name}</span>
        </span>
        <ChevronDownIcon className={`w-5 h-5 text-gray-400 transition-transform ${isOpen ? 'rotate-180' : ''}`} />
      </button>

      {isOpen && (
        <div className="absolute z-10 w-full mt-1 bg-white border border-gray-300 rounded-md shadow-lg">
          <div className="py-1">
            {workspaces.map((workspace) => (
              <button
                key={workspace.id}
                onClick={() => handleWorkspaceSelect(workspace)}
                className={`flex items-center w-full px-4 py-2 text-left hover:bg-gray-100 ${
                  workspace.id === currentWorkspace?.id ? 'bg-blue-50 text-blue-700' : 'text-gray-900'
                }`}
              >
                <div className="w-6 h-6 bg-gray-300 rounded-full flex items-center justify-center text-gray-700 text-xs font-medium mr-3">
                  {workspace.name.charAt(0).toUpperCase()}
                </div>
                <div className="flex-1">
                  <div className="font-medium">{workspace.name}</div>
                  <div className="text-xs text-gray-500">{workspace.role}</div>
                </div>
              </button>
            ))}
          </div>
        </div>
      )}
    </div>
  );
};
Custom Hook for Data Fetching: A reusable hook for API data fetching with loading, error, and retry states.
useApi.ts 
import { useState, useEffect, useCallback, useRef } from 'react';

interface ApiState<T> {
  data: T | null;
  loading: boolean;
  error: Error | null;
  refetch: () => Promise;
}

interface UseApiOptions {
  enabled?: boolean;
  retryCount?: number;
  retryDelay?: number;
  onSuccess?: (data: any) => void;
  onError?: (error: Error) => void;
}

export function useApi<T>(
  url: string,
  options: UseApiOptions = {}
): ApiState<T> {
  const {
    enabled = true,
    retryCount = 3,
    retryDelay = 1000,
    onSuccess,
    onError
  } = options;

  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const abortControllerRef = useRef<AbortController | null>(null);
  const retryTimeoutRef = useRef<NodeJS.Timeout | null>(null);

  const fetchData = useCallback(async () => {
    if (!enabled) return;

    // Cancel any ongoing request
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
    }

    // Clear any retry timeout
    if (retryTimeoutRef.current) {
      clearTimeout(retryTimeoutRef.current);
    }

    abortControllerRef.current = new AbortController();
    setLoading(true);
    setError(null);

    try {
      const response = await fetch(url, {
        signal: abortControllerRef.current.signal,
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('access_token')}`
        }
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const result = await response.json();
      setData(result);
      onSuccess?.(result);
    } catch (err) {
      if (err instanceof Error && err.name === 'AbortError') {
        return; // Request was cancelled, don't update state
      }

      const error = err instanceof Error ? err : new Error('Unknown error');
      setError(error);
      onError?.(error);
    } finally {
      setLoading(false);
    }
  }, [url, enabled, onSuccess, onError]);

  const refetch = useCallback(async () => {
    await fetchData();
  }, [fetchData]);

  // Retry logic
  useEffect(() => {
    if (error && retryCount > 0) {
      retryTimeoutRef.current = setTimeout(() => {
        fetchData();
      }, retryDelay);
    }

    return () => {
      if (retryTimeoutRef.current) {
        clearTimeout(retryTimeoutRef.current);
      }
    };
  }, [error, retryCount, retryDelay, fetchData]);

  // Initial fetch
  useEffect(() => {
    fetchData();

    return () => {
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }
      if (retryTimeoutRef.current) {
        clearTimeout(retryTimeoutRef.current);
      }
    };
  }, [fetchData]);

  return { data, loading, error, refetch };
}

Authentication Middleware: JWT authentication middleware with role-based access control and workspace isolation.
auth_middleware.py 
from typing import Optional, List
from fastapi import HTTPException, Depends, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import jwt, JWTError
from sqlalchemy.orm import Session
from datetime import datetime, timezone

from .database import get_db
from .models import User, Workspace, WorkspaceMember
from .config import settings

security = HTTPBearer()

class AuthService:
    def __init__(self):
        self.secret_key = settings.SECRET_KEY
        self.algorithm = settings.ALGORITHM
        self.access_token_expire_minutes = settings.ACCESS_TOKEN_EXPIRE_MINUTES

    def create_access_token(self, data: dict) -> str:
        to_encode = data.copy()
        expire = datetime.now(timezone.utc) + timedelta(minutes=self.access_token_expire_minutes)
        to_encode.update({"exp": expire})
        return jwt.encode(to_encode, self.secret_key, algorithm=self.algorithm)

    def verify_token(self, token: str) -> dict:
        try:
            payload = jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
            return payload
        except JWTError:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Could not validate credentials",
                headers={"WWW-Authenticate": "Bearer"},
            )

    async def get_current_user(
        self,
        credentials: HTTPAuthorizationCredentials = Depends(security),
        db: Session = Depends(get_db)
    ) -> User:
        token_data = self.verify_token(credentials.credentials)
        user_id = token_data.get("sub")
        
        if not user_id:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid authentication credentials"
            )

        user = db.query(User).filter(User.id == user_id).first()
        if not user:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="User not found"
            )

        return user

    async def get_current_active_user(
        self,
        current_user: User = Depends(get_current_user)
    ) -> User:
        if not current_user.is_active:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Inactive user"
            )
        return current_user

    async def get_workspace_member(
        self,
        workspace_id: str,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db)
    ) -> WorkspaceMember:
        workspace_member = db.query(WorkspaceMember).filter(
            WorkspaceMember.workspace_id == workspace_id,
            WorkspaceMember.user_id == current_user.id
        ).first()

        if not workspace_member:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Access denied to this workspace"
            )

        return workspace_member

    def require_roles(self, required_roles: List[str]):
        def role_checker(
            current_user: User = Depends(self.get_current_active_user),
            workspace_member: WorkspaceMember = Depends(self.get_workspace_member)
        ):
            if workspace_member.role not in required_roles:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail=f"This action requires one of these roles: {', '.join(required_roles)}"
                )
            return current_user
        return role_checker

# Usage examples
def require_admin():
    return AuthService().require_roles(['owner', 'admin'])

def require_owner():
    return AuthService().require_roles(['owner'])

# Dependency shortcuts
current_user = AuthService().get_current_active_user
admin_required = require_admin()
owner_required = require_owner()
Background Task Queue: Redis-based task queue for handling background jobs with retry logic and monitoring.
task_queue.py 
import json
import asyncio
import logging
from typing import Dict, Any, Optional, Callable
from datetime import datetime, timedelta
import redis.asyncio as redis
from celery import Celery
from celery.utils.log import get_task_logger

from .config import settings
from .database import get_db_session
from .models import Task, TaskStatus

logger = get_task_logger(__name__)

class TaskQueue:
    def __init__(self):
        self.redis_client = redis.from_url(settings.REDIS_URL)
        self.celery_app = Celery(
            'webcraft',
            broker=settings.REDIS_URL,
            backend=settings.REDIS_URL
        )
        self.celery_app.conf.update(
            task_serializer='json',
            accept_content=['json'],
            result_serializer='json',
            timezone='UTC',
            enable_utc=True,
            task_track_started=True,
            task_time_limit=30 * 60,  # 30 minutes
            task_soft_time_limit=25 * 60,  # 25 minutes
            worker_prefetch_multiplier=1,
            worker_max_tasks_per_child=1000,
        )

    async def enqueue_task(
        self,
        task_name: str,
        task_data: Dict[str, Any],
        priority: int = 5,
        delay: Optional[int] = None
    ) -> str:
        """Add a task to the queue with optional delay"""
        task_id = f"{task_name}_{datetime.now().timestamp()}"
        
        # Store task metadata in Redis
        task_metadata = {
            'task_id': task_id,
            'task_name': task_name,
            'task_data': task_data,
            'priority': priority,
            'created_at': datetime.now().isoformat(),
            'status': 'pending'
        }
        
        await self.redis_client.hset(
            f"task:{task_id}",
            mapping=task_metadata
        )

        # Schedule the task
        if delay:
            # Use delayed task execution
            self.celery_app.send_task(
                task_name,
                args=[task_data],
                task_id=task_id,
                countdown=delay
            )
        else:
            # Execute immediately
            self.celery_app.send_task(
                task_name,
                args=[task_data],
                task_id=task_id,
                priority=priority
            )

        logger.info(f"Task {task_id} enqueued with priority {priority}")
        return task_id

    async def get_task_status(self, task_id: str) -> Dict[str, Any]:
        """Get the current status of a task"""
        task_data = await self.redis_client.hgetall(f"task:{task_id}")
        
        if not task_data:
            return {'status': 'not_found'}

        # Get Celery task result if available
        async_result = self.celery_app.AsyncResult(task_id)
        
        return {
            'task_id': task_id,
            'status': async_result.status,
            'result': async_result.result if async_result.successful() else None,
            'error': str(async_result.result) if async_result.failed() else None,
            'created_at': task_data.get('created_at'),
            'started_at': task_data.get('started_at'),
            'completed_at': task_data.get('completed_at')
        }

    async def cancel_task(self, task_id: str) -> bool:
        """Cancel a pending task"""
        async_result = self.celery_app.AsyncResult(task_id)
        
        if async_result.state in ['PENDING', 'RETRY']:
            async_result.revoke()
            await self.redis_client.hset(f"task:{task_id}", 'status', 'cancelled')
            logger.info(f"Task {task_id} cancelled")
            return True
        
        return False

        send_invitation_email(invitation)
        
        # Update invitation status

        asyncio.run(queue.redis_client.hset(
            f"task:{task_id}",
            'status': 'completed',
            'completed_at': datetime.now().isoformat()
        ))

        logger.info(f"Workspace invitation task {task_id} completed successfully")
        return {"status": "success", "invitation_id": invitation.id}

    except Exception as exc:
        logger.error(f"Workspace invitation task {task_id} failed: {str(exc)}")
        
        # Retry logic
        if self.request.retries < self.max_retries:
            logger.warning(f"Retrying task {task_id} (attempt {self.request.retries + 1})")
            raise self.retry(countdown=60 * (self.request.retries + 1))
        
        # Mark task as failed
        queue = TaskQueue()
        asyncio.run(queue.redis_client.hset(
            f"task:{task_id}",
            'status': 'failed',
            'error': str(exc),
            'completed_at': datetime.now().isoformat()
        ))
        
        raise exc

-- Set workspace context for RLS
SET app.current_workspace_id = 'ws_123e4567-e89b-12d3-a456-426614174000';

-- Get projects with member counts and recent activity
WITH project_stats AS (
    SELECT 
        p.id,
        p.name,
        p.description,
        p.created_at,
        p.updated_at,
        COUNT(DISTINCT pm.user_id) as member_count,
        COUNT(DISTINCT t.id) as task_count,
        MAX(t.updated_at) as last_activity,
        SUM(CASE WHEN t.status = 'completed' THEN 1 ELSE 0 END) as completed_tasks,
        SUM(CASE WHEN t.status = 'in_progress' THEN 1 ELSE 0 END) as in_progress_tasks
    FROM projects p
    LEFT JOIN project_members pm ON p.id = pm.project_id
    LEFT JOIN tasks t ON p.id = t.project_id
    WHERE p.workspace_id = current_setting('app.current_workspace_id')::UUID
    GROUP BY p.id, p.name, p.description, p.created_at, p.updated_at
),
user_role AS (
    SELECT 
        pm.project_id,
        pm.role,
        u.name as user_name,
        u.email as user_email
    FROM project_members pm
    JOIN users u ON pm.user_id = u.id
    WHERE pm.user_id = current_setting('app.current_user_id')::UUID
)
SELECT 
    ps.*,
    ur.role as user_role,
    CASE 
        WHEN ps.last_activity > NOW() - INTERVAL '7 days' THEN 'active'
        WHEN ps.last_activity > NOW() - INTERVAL '30 days' THEN 'recent'
        ELSE 'inactive'
    END as activity_status,
    ROUND(
        (ps.completed_tasks::DECIMAL / NULLIF(ps.task_count, 0)) * 100, 
        2
    ) as completion_rate
FROM project_stats ps
LEFT JOIN user_role ur ON ps.id = ur.project_id
ORDER BY ps.last_activity DESC NULLS LAST, ps.created_at DESC;

Time-series Data with TimescaleDB: Efficient queries for time-series data using TimescaleDB hypertables.

-- Create hypertable for metrics
CREATE TABLE metrics (
    time TIMESTAMPTZ NOT NULL,
    workspace_id UUID NOT NULL,
    metric_name TEXT NOT NULL,
    metric_value DOUBLE PRECISION,
    tags JSONB
);

-- Convert to hypertable
SELECT create_hypertable('metrics', 'time', 
    chunk_time_interval => INTERVAL '1 day');

-- Create compression policy
SELECT add_compression_policy('metrics', INTERVAL '7 days');

-- Enable compression
ALTER TABLE metrics SET (
    timescaledb.compress,
    timescaledb.compress_orderby = 'time DESC',
    timescaledb.compress_segmentby = 'workspace_id, metric_name'
);

-- Continuous aggregate for hourly metrics
CREATE MATERIALIZED VIEW metrics_hourly
WITH (timescaledb.continuous) AS
SELECT 
    time_bucket('1 hour', time) as hour,
    workspace_id,
    metric_name,
    AVG(metric_value) as avg_value,
    MIN(metric_value) as min_value,
    MAX(metric_value) as max_value,
    COUNT(*) as sample_count
FROM metrics
GROUP BY hour, workspace_id, metric_name
WITH NO DATA;

-- Refresh policy for continuous aggregate
SELECT add_continuous_aggregate_policy('metrics_hourly',
    start_offset => INTERVAL '1 month',
    end_offset => INTERVAL '1 hour',
    schedule_interval => INTERVAL '1 hour');

-- Real-time analytics query
WITH hourly_stats AS (
    SELECT 
        time_bucket('1 hour', m.time) as hour,
        m.workspace_id,
        COUNT(DISTINCT m.metric_name) as unique_metrics,
        AVG(m.metric_value) as avg_metric_value,
        SUM(CASE WHEN m.metric_value > 0 THEN 1 ELSE 0 END) as positive_metrics
    FROM metrics m
    WHERE m.time > NOW() - INTERVAL '24 hours'
    AND m.workspace_id = $1
    GROUP BY hour, m.workspace_id
),
daily_trends AS (
    SELECT 
        time_bucket('1 day', m.time) as day,
        m.workspace_id,
        metric_name,
        AVG(metric_value) as daily_avg,
        LAG(AVG(metric_value)) OVER (PARTITION BY m.workspace_id, m.metric_name ORDER BY time_bucket('1 day', m.time)) as previous_day_avg,
        (AVG(metric_value) - LAG(AVG(metric_value)) OVER (PARTITION BY m.workspace_id, m.metric_name ORDER BY time_bucket('1 day', m.time))) / NULLIF(LAG(AVG(metric_value)) OVER (PARTITION BY m.workspace_id, m.metric_name ORDER BY time_bucket('1 day', m.time)), 0) * 100 as day_over_day_change
    FROM metrics m
    WHERE m.time > NOW() - INTERVAL '7 days'
    AND m.workspace_id = $1
    GROUP BY day, m.workspace_id, m.metric_name
)
SELECT 
    h.*,
    d.metric_name,
    d.daily_avg,
    d.day_over_day_change,
    CASE 
        WHEN d.day_over_day_change > 10 THEN 'increasing'
        WHEN d.day_over_day_change < -10 THEN 'decreasing'
        ELSE 'stable'
    END as trend_direction
FROM hourly_stats h
JOIN daily_trends d ON h.workspace_id = d.workspace_id
ORDER BY h.hour DESC, d.metric_name;

-- Performance optimization for large datasets
-- Create index on compressed chunks
CREATE INDEX idx_metrics_compressed_workspace_time 
ON metrics(workspace_id, time DESC) 
WHERE metric_value IS NOT NULL;

GitHub Actions CI/CD Pipeline: Complete CI/CD pipeline with testing, building, and deployment stages.
.github/workflows/deploy.yml 
name: Deploy to Production

on:
  push:
    branches: [main]
  workflow_dispatch:

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  test:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: webcraft_test
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432
      redis:
        image: redis:7-alpine
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 6379:6379

    steps:
    - name: Checkout code
      uses: actions/checkout@v4

    - name: Setup Node.js
      uses: actions/setup-node@v4
      with:
        node-version: '18'
        cache: 'yarn'

    - name: Setup Python
      uses: actions/setup-python@v4
      with:
        python-version: '3.11'
        cache: 'pip'

    - name: Install dependencies
      run: |
        yarn install --frozen-lockfile
        pip install -r requirements.txt
        pip install -r requirements-test.txt

    - name: Run linting
      run: |
        yarn lint
        yarn type-check
        flake8 apps/api
        black --check apps/api

    - name: Run frontend tests
      run: yarn test --ci --coverage --watchAll=false
      env:
        CI: true

    - name: Run backend tests
      run: |
        pytest apps/api/tests/ -v --cov=apps/api --cov-report=xml
      env:
        DATABASE_URL: postgresql://postgres:postgres@localhost:5432/webcraft_test
        REDIS_URL: redis://localhost:6379/1
        TESTING: true

    - name: Upload coverage reports
      uses: codecov/codecov-action@v3
      with:
        files: ./coverage/coverage-final.json,./coverage.xml
        flags: unittests
        name: codecov-umbrella

  build:
    needs: test
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write

    steps:
    - name: Checkout code
      uses: actions/checkout@v4

    - name: Setup Docker Buildx
      uses: docker/setup-buildx-action@v3

    - name: Log in to Container Registry
      uses: docker/login-action@v3
      with:
        registry: ${{ env.REGISTRY }}
        username: ${{ github.actor }}
        password: ${{ secrets.GITHUB_TOKEN }}

    - name: Extract metadata
      id: meta
      uses: docker/metadata-action@v5
      with:
        images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}
        tags: |
          type=ref,event=branch
          type=ref,event=pr
          type=sha,prefix={{branch}}-
          type=raw,value=latest,enable={{is_default_branch}}

    - name: Build and push Docker image
      uses: docker/build-push-action@v5
      with:
        context: .
        push: true
        tags: ${{ steps.meta.outputs.tags }}
        labels: ${{ steps.meta.outputs.labels }}
        cache-from: type=gha
        cache-to: type=gha,mode=max
        platforms: linux/amd64,linux/arm64

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment: production
    if: github.ref == 'refs/heads/main'

    steps:
    - name: Checkout code
      uses: actions/checkout@v4

    - name: Setup kubectl
      uses: azure/setup-kubectl@v3
      with:
        version: 'v1.28.0'

    - name: Setup Helm
      uses: azure/setup-helm@v3
      with:
        version: 'v3.12.0'

    - name: Configure AWS credentials
      uses: aws-actions/configure-aws-credentials@v4
      with:
        aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
        aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
        aws-region: us-west-2

    - name: Update kubeconfig
      run: |
        aws eks update-kubeconfig --region us-west-2 --name webcraft-prod

    - name: Deploy with Helm
      run: |
        helm upgrade --install webcraft ./helm/webcraft \
          --namespace webcraft \
          --create-namespace \
          --set image.tag=${{ github.sha }} \
          --set environment=production \
          --wait --timeout=10m

    - name: Verify deployment
      run: |
        kubectl rollout status deployment/webcraft-frontend -n webcraft
        kubectl rollout status deployment/webcraft-api -n webcraft

    - name: Run smoke tests
      run: |
        kubectl run smoke-test --image=curlimages/curl:latest --rm -i --restart=Never \
          --command -- curl -f http://webcraft-api-service.webcraft.svc.cluster.local/health

    - name: Notify deployment status
      if: always()
      uses: 8398a7/action-slack@v3
      with:
        status: ${{ job.status }}
        channel: '#deployments'
        webhook_url: ${{ secrets.SLACK_WEBHOOK_URL }}
        fields: repo,message,commit,author,action,eventName,ref,workflow
