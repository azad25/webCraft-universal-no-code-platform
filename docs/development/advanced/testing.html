Testing Strategies

Comprehensive testing approaches for WebCraft applications

Testing Overview

WebCraft employs a comprehensive testing strategy covering unit tests, integration tests, end-to-end tests, and performance tests. Our testing approach follows the testing pyramid principle with emphasis on fast, reliable unit tests.

Testing Stack

  • Frontend: Jest, React Testing Library, Cypress, Playwright
  • Backend: pytest, pytest-asyncio, FastAPI TestClient
  • Database: pytest-postgresql, testcontainers
  • Performance: Locust, k6, Artillery
  • Coverage: Coverage.py, Istanbul/nyc

import { renderHook, act } from '@testing-library/react-hooks';
import { useForm } from './useForm';

describe('useForm Hook', () => {
  it('initializes with default values', () => {
    const { result } = renderHook(() => useForm({
      initialValues: { name: 'John', email: '' }
    }));
    
    expect(result.current.values).toEqual({ name: 'John', email: '' });
  });

  it('updates values on change', () => {
    const { result } = renderHook(() => useForm());
    
    act(() => {
      result.current.handleChange('name', 'Jane');
    });
    
    expect(result.current.values.name).toBe('Jane');
  });

  it('validates form on submit', async () => {
    const onSubmit = jest.fn();
    const { result } = renderHook(() => useForm({
      onSubmit,
      validate: (values) => {
        if (!values.email) return { email: 'Email is required' };
        return {};
      }
    }));
    
    await act(async () => {
      await result.current.handleSubmit();
    });
    
    expect(result.current.errors.email).toBe('Email is required');
    expect(onSubmit).not.toHaveBeenCalled();
  });
});

# performance_monitor.py
import time
import psutil
import asyncio
from functools import wraps
from typing import Callable, Any

def measure_performance(func: Callable) -> Callable:
    """Decorator to measure function performance"""
    @wraps(func)
    async def async_wrapper(*args, **kwargs):
        start_time = time.time()
        start_memory = psutil.Process().memory_info().rss / 1024 / 1024  # MB
        
        try:
            result = await func(*args, **kwargs)
            
            end_time = time.time()
            end_memory = psutil.Process().memory_info().rss / 1024 / 1024
            
            execution_time = end_time - start_time
            memory_usage = end_memory - start_memory
            
            print(f"{func.__name__} - Time: {execution_time:.3f}s, Memory: {memory_usage:.2f}MB")
            
            return result
        except Exception as e:
            print(f"{func.__name__} - Error: {str(e)}")
            raise
    
    @wraps(func)
    def sync_wrapper(*args, **kwargs):
        start_time = time.time()
        start_memory = psutil.Process().memory_info().rss / 1024 / 1024
        
        try:
            result = func(*args, **kwargs)
            
            end_time = time.time()
            end_memory = psutil.Process().memory_info().rss / 1024 / 1024
            
            execution_time = end_time - start_time
            memory_usage = end_memory - start_memory
            
            print(f"{func.__name__} - Time: {execution_time:.3f}s, Memory: {memory_usage:.2f}MB")
            
            return result
        except Exception as e:
            print(f"{func.__name__} - Error: {str(e)}")
            raise
    
    return async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper

                "name": row.project_name
            })
    
    workspace_data["projects"] = projects
    return workspace_data

async def test_database_performance():
    """Test database query performance"""
    import asyncpg
    
    # Connection pool performance
    pool = await asyncpg.create_pool(
        "postgresql://user:pass@localhost/webcraft",
        min_size=10,
        max_size=20
    )
    
    # Test query performance
    queries = [
        ("Simple select", "SELECT * FROM users LIMIT 100"),
        ("Complex join", """
            SELECT u.*, w.name as workspace_name, COUNT(p.id) as project_count
            FROM users u
            JOIN workspace_members wm ON u.id = wm.user_id
            JOIN workspaces w ON wm.workspace_id = w.id
            LEFT JOIN projects p ON w.id = p.workspace_id
            GROUP BY u.id, w.name
            LIMIT 100
        """),
        ("Aggregation", """
            SELECT 
                DATE(created_at) as date,
                COUNT(*) as user_count,
                COUNT(CASE WHEN is_active THEN 1 END) as active_count
            FROM users
            WHERE created_at >= CURRENT_DATE - INTERVAL '30 days'
            GROUP BY DATE(created_at)
            ORDER BY date DESC
        """),
    ]
    
    async with pool.acquire() as conn:
        for name, query in queries:
            start_time = time.time()
            
            # Execute query multiple times
            for _ in range(10):
                await conn.fetch(query)
            
            avg_time = (time.time() - start_time) / 10
            print(f"{name}: {avg_time:.3f}s average")

class TestDataBuilder:
    """Builder pattern for complex test data"""
    
    def __init__(self):
        self.workspace = None
        self.projects = []
        self.pages = []
        self.users = []
    
    def with_workspace(self, **kwargs):
        self.workspace = WorkspaceFactory(**kwargs)
        return self
    
    def with_projects(self, count=1, **kwargs):
        for _ in range(count):
            project = ProjectFactory(workspace=self.workspace, **kwargs)
            self.projects.append(project)
        return self
    
    def with_pages(self, count=1, **kwargs):
        for project in self.projects:
            for _ in range(count):
                page = PageFactory(project=project, **kwargs)
                self.pages.append(page)
        return self
    
    def with_users(self, count=1, **kwargs):
        for _ in range(count):
            user = UserFactory(**kwargs)
            self.users.append(user)
        return self
    
    def build(self):
        return {
            'workspace': self.workspace,
            'projects': self.projects,
            'pages': self.pages,
            'users': self.users
        }

Testing Best Practices

Test Organization

  • Arrange-Act-Assert (AAA): Structure tests clearly
  • One assertion per test: Keep tests focused and simple
  • Descriptive test names: Test names should explain what is being tested
  • Test isolation: Tests should not depend on each other
  • Use fixtures: Share common test setup

  • Aim for 80%+ code coverage

  • Test edge cases and error conditions
  • Don't write tests just to increase coverage
  • Use coverage reports to identify untested code
